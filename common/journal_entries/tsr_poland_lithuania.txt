je_odrodzenie = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	
	group = je_group_tsr_content

	is_shown_when_inactive = {
		c:POL ?= this
		NOR = {
			is_revolutionary = yes
			is_secessionist = yes
			is_subject = yes
		}
	}	
	
	scripted_button = je_odrodzenie_polish_radicals_subject_button
	scripted_button = je_odrodzenie_polish_radicals_overlord_button

	immediate = {
		trigger_event = { id = poland_lithuania_events.4 popup = yes }
	}
		
	possible = {
		has_technology_researched = nationalism
	}

	complete = {
		any_state = {
			filter = { 
				region = sr:region_poland 
				state_region = {
					is_homeland = cu:polish
				}
			}
			count = all
			OR = {
				owner = root
				owner = { is_subject_of = root }
			}
		}
	}

	on_complete = {
		trigger_event = { id = poland_events.4 popup = yes }
		add_journal_entry = { type = je_greater_poland }
	}
	
	fail = {
		OR = {
			is_subject = yes
			country_rank <= rank_value:minor_power
		}
	}
	
	invalid = {
		NOT = { exists = c:POL }
	}

	weight = 100

	should_be_pinned_by_default = yes
}
