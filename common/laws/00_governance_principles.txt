law_chiefdom = {
	group = lawgroup_governance_principles
	
	icon = "gfx/interface/icons/law_icons/chiefdom.dds"
	
	progressiveness = -100
	
	can_enact = {
		is_country_type = decentralized
	}
	
	is_visible = {
		is_country_type = decentralized
	}
	
	modifier = {
		country_legitimacy_headofstate_add = 20
		country_authority_add = 100
		state_education_access_add = -0.2
		country_weekly_innovation_add = -5
		country_tech_spread_add = -10
	}
	
	pop_support = {
		value = 0
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = {
				law_is_important = yes
			}
			add = 750
		}
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
}

law_presidential_republic = {
	group = lawgroup_governance_principles
	
	icon = "gfx/interface/icons/law_icons/presidential_republic.dds"
	
	progressiveness = 100
	
	disallowing_laws = {
		law_governorate
	}
	
	unlocking_technologies = {
		democracy
	}
	
	modifier = {
		country_legitimacy_headofstate_add = 10
		country_legitimacy_govt_total_votes_add = 10
		country_legitimacy_ideological_incoherence_mult = 0.2
		country_authority_add = 100
	}
	
	on_activate = {
		if = {
			limit = {
				ig:ig_landowners = {
					OR = {
						has_ideology = ideology:ideology_paternalistic
						has_ideology = ideology:ideology_scholar_paternalistic
						has_ideology = ideology:ideology_junker_paternalistic
						has_ideology = ideology:ideology_papal_paternalistic
						has_ideology = ideology:ideology_safavid_paternalistic
						has_ideology = ideology:ideology_akkovin_paternalistic
					}
				}
			}
			ig:ig_landowners = {
				remove_ideology = ideology_paternalistic
				remove_ideology = ideology_scholar_paternalistic
				remove_ideology = ideology_junker_paternalistic
				remove_ideology = ideology_papal_paternalistic
				remove_ideology = ideology_safavid_paternalistic
				remove_ideology = ideology_akkovin_paternalistic
				add_ideology = ideology_republican_paternalistic
			}
		}
		if = {
			limit = {
				any_interest_group = {
					has_ideology = ideology:ideology_constitutionalist
				}
			}
			every_interest_group = {
				limit = {
					has_ideology = ideology:ideology_constitutionalist
				}
				remove_ideology = ideology_constitutionalist
				add_ideology = ideology_republican
			}
		}
	}
	
	possible_political_movements = {
		law_council_republic
		law_monarchy
	}

	pop_support = {
		value = 0
		add = {
			desc = "POP_MIDDLE_STRATA"
			if = {
				limit = { 
					strata = middle
				}
				value = 0.05
			}
		}
		add = {
            desc = "POP_LITERACY"
            
                if = {
                    limit = {
                        wealth < 15
                    }
                    add = {
                        value = literacy_rate
                        divide = 2
                    }
                }
                if = {
                    limit = {
                        wealth < 10
                    }
                    add = {
                        value = literacy_rate
                        divide = 4
                    }
					multiply = literacy_rate
                }		
        }
		multiply = {
			desc = "TUR_NO_BALKAN_REVOLTS_OVER_LAWS"			
			if = {
				limit = {
					owner = { 
						c:TUR ?= this
					}
					NOT = { pop_is_discriminated = yes }
				}
				value = 0
			}
		}
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			has_ideology = ideology:ideology_republican_leader
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = {
				law_is_important = yes
			}
			add = 750
		}
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
}
