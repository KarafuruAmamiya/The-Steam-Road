balkan_crisis.1 = {
	type = country_event
	placement = root
	event_image = {
		video = "middleeast_police_breaking_door"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_revolt.dds"

	title = balkan_crisis.1.t
	desc = balkan_crisis.1.d
	flavor = balkan_crisis.1.f

	duration = 3

	trigger = {
		#triggered by JE
	}

	immediate = {
	}

	option = {
		name = balkan_crisis.1.a
		default_option = yes
		
		hidden_effect = {
			every_country = {
				limit = {
					AND = {
						OR = {
							has_diplomatic_relevance = c:TUR
							has_interest_marker_in_region = region_greece
							has_interest_marker_in_region = region_balkans
						}
						NOT = { THIS = ROOT }
						OR = {
							country_rank >= rank_value:major_power
							is_subject_of = c:TUR
							trigger_if = {
								limit = {
									OR = {
										c:POL ?= this
										c:PLC ?= this
									}
								}
								OR = {
									any_neighbouring_state = {
										owner = {
											OR = {
												c:TUR ?= this
												is_subject_of = c:TUR
											}
										}
									}
									any_scope_state = {
										has_claim_by = c:TUR
									}
									any_subject_or_below = {
										any_scope_state = {
											has_claim_by = c:TUR
										}
									}
								}
							}
						}
					}
				}
				trigger_event = { id = balkan_crisis.2 days = 7 }
			}
		}
		
		ai_chance = {
			base = 100
		}
	}
}

balkan_crisis.2 = {
	type = country_event
	placement = root
	event_image = {
		video = "middleeast_police_breaking_door"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_revolt.dds"

	title = balkan_crisis.2.t
	desc = balkan_crisis.2.d
	flavor = balkan_crisis.2.f

	duration = 3

	trigger = {
		#triggered by JE
	}

	immediate = {
		c:TUR ?= { save_scope_as = tur_scope }
	}

	option = { # Intervene in the Balkans
		name = balkan_crisis.2.a
		
		add_journal_entry = {
			type = je_balkan_crisis_display
		}
		
		change_relations = {
			country = c:TUR
			value = -30
		}
		
		hidden_effect = {
			if = {
				limit = {
					NOT = { is_subject_of = c:TUR }
				}
				ROOT = { save_scope_as = balkan_intervention_scope }
				every_country = {
					limit = {
						OR = {
							c:TUR ?= this
							has_diplomatic_relevance = c:TUR
							has_interest_marker_in_region = region_greece
							has_interest_marker_in_region = region_balkans
						}
					}
					post_notification = country_intervened_in_balkans_crisis_notification
				}
			}
		}
		
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					c:TUR ?= {
						relations:root <= relations_threshold:neutral
					}
				}
				add = 100
			}
			modifier = {
				trigger = {
					country_rank >= rank_value:great_power 
				}
				add = 100
			}
			modifier = {
				trigger = {
					has_diplomatic_pact = {
						who = c:TUR
						type = rivalry
					}
				}
				add = 1000
			}
			# Persia and Akkovins want nothing to do with Balkans
			modifier = {
				trigger = {
					c:PER ?= this
				}
				add = -10000
			}
			modifier = {
				trigger = {
					c:AKN ?= this
					NOT = { is_subject_of = c:TUR }
				}
				add = -10000
			}
			modifier = {
				trigger = {
					OR = {
						c:POL ?= {
							root = this
							country_or_subject_owns_entire_state_region = STATE_EAST_GALICIA
						}
						c:PLC ?= {
							root = this
							country_or_subject_owns_entire_state_region = STATE_EAST_GALICIA
						}
					}
				}
				add = 1000
			}
		}
	}
	
	option = { # Let's not get involved
		name = balkan_crisis.2.b
		default_option = yes
		trigger = {
			NOT = { is_subject_of = c:TUR }
		}
		
		ai_chance = {
			base = 50
			# Persia and Akkovins want nothing to do with Balkans
			modifier = {
				trigger = {
					c:PER ?= this
				}
				add = 10000
			}
			modifier = {
				trigger = {
					c:AKN ?= this
					NOT = { is_subject_of = c:TUR }
				}
				add = 10000
			}
		}
	}
}

# Rising Tension in Balkan State
balkan_crisis.3 = {
	type = country_event
	placement = scope:crisis_state
	title = balkan_crisis.3.t
	desc = balkan_crisis.3.d
	flavor =  balkan_crisis.3.f
	
	duration = 1
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	
	event_image = {
		video = "middleeast_police_breaking_door"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		NOT = { has_variable = uprising_cooldown }
		NOT = { has_variable = massacre_happened }
		OR = {
			var:greek_states_radicals_total_final >= 50
			var:serb_states_radicals_total_final >= 50
			var:croat_states_radicals_total_final >= 50
			var:bulgarian_states_radicals_total_final >= 50
		}
	}
	immediate = {
		set_variable = { name = uprising_cooldown months = 12 }
		if = {
			limit = {
				var:greek_states_radicals_total_final > var:serb_states_radicals_total_final
				var:greek_states_radicals_total_final > var:croat_states_radicals_total_final
				var:greek_states_radicals_total_final > var:bulgarian_states_radicals_total_final
				NOT = {
					any_scope_state = {
						state_region = {
							is_homeland = cu:greek
						}
						count >= 4
						has_modifier = quelling_uprising
					}
				}
			}
			set_variable = greek_uprising
			cu:greek = { save_scope_as = crisis_culture }
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:greek
					}
				}
				save_scope_as = crisis_state
				set_variable = crisis_state_var
			}
		}
		else_if = {
			limit = {
				var:serb_states_radicals_total_final > var:greek_states_radicals_total_final
				var:serb_states_radicals_total_final > var:croat_states_radicals_total_final
				var:serb_states_radicals_total_final > var:bulgarian_states_radicals_total_final
				NOT = {
					any_scope_state = {
						state_region = {
							is_homeland = cu:serb
						}
						count >= 4
						has_modifier = quelling_uprising
					}
				}
			}
			set_variable = serb_uprising
			cu:serb = { save_scope_as = crisis_culture }
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:serb
					}
				}
				save_scope_as = crisis_state
				set_variable = crisis_state_var
			}
		}
		else_if = {
			limit = {
				var:croat_states_radicals_total_final > var:greek_states_radicals_total_final
				var:croat_states_radicals_total_final > var:serb_states_radicals_total_final
				var:croat_states_radicals_total_final > var:bulgarian_states_radicals_total_final
				NOT = {
					any_scope_state = {
						state_region = {
							is_homeland = cu:croat
						}
						count >= 3
						has_modifier = quelling_uprising
					}
				}
			}
			set_variable = croat_uprising
			cu:croat = { save_scope_as = crisis_culture }
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:croat
					}
				}
				save_scope_as = crisis_state
				set_variable = crisis_state_var
			}
		}
		else_if = {
			limit = {
				var:bulgarian_states_radicals_total_final > var:greek_states_radicals_total_final
				var:bulgarian_states_radicals_total_final > var:serb_states_radicals_total_final
				var:bulgarian_states_radicals_total_final > var:croat_states_radicals_total_final
				NOT = {
					any_scope_state = {
						state_region = {
							is_homeland = cu:bulgarian
						}
						count >= 4
						has_modifier = quelling_uprising
					}
				}
			}
			set_variable = bulgarian_uprising
			cu:bulgarian = { save_scope_as = crisis_culture }
			random_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:bulgarian
					}
				}
				save_scope_as = crisis_state
				set_variable = crisis_state_var
			}
		}
	}
	option = { #Let's hope it doesn't get too bad
		name = balkan_crisis.3.a
		default_option = yes
		hidden_effect = {
			trigger_event = {
				id = balkan_crisis.4
				days = 30
			}
		}
		custom_tooltip = {
			text = balkan_crisis_progress_increase_10_tt
			change_global_variable = { name = balkan_crisis_main_var add = 10 }
		}
		scope:crisis_state = {
			add_modifier = {
				name = balkan_uprising
				months = 6
			}
			add_radicals_in_state = {
				value = large_radicals
				culture = scope:crisis_culture
			}
		}
	}
}

# Regional crisis escalates
balkan_crisis.4 = {
	type = country_event
	placement = scope:crisis_state
	title = balkan_crisis.4.t
	desc = balkan_crisis.4.d
	flavor =  balkan_crisis.4.f
	
	duration = 1
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	
	event_image = {
		video = "middleeast_police_breaking_door"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	immediate = {
		random_scope_state = {
			limit = { has_variable = crisis_state_var }
			save_scope_as = crisis_state
		}
		if = {
			limit = { has_variable = greek_uprising }
			cu:greek = { save_scope_as = crisis_culture }
		}
		if = {
			limit = { has_variable = serb_uprising }
			cu:serb = { save_scope_as = crisis_culture }
		}
		if = {
			limit = { has_variable = croat_uprising }
			cu:croat = { save_scope_as = crisis_culture }
		}
		if = {
			limit = { has_variable = bulgarian_uprising }
			cu:bulgarian = { save_scope_as = crisis_culture }
		}
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = scope:crisis_culture
				}
				NOT = {
					scope:crisis_state ?= this
				}
			}
			add_modifier = {
				name = balkan_uprising
				months = 6
			}
			add_radicals_in_state = {
				value = large_radicals
				culture = scope:crisis_culture
			}
		}
	}
	trigger = {
		#triggered by prior event
	}
	option = { #Rely on the Bashi-bazouk
		name = balkan_crisis.4.a
		default_option = yes
		scope:crisis_state.state_region = {
			add_devastation = 20
		}
		scope:crisis_state = {
			add_radicals_in_state = {
				value = very_large_radicals
				culture = scope:crisis_culture
			}
		}
		hidden_effect = {
			random_list = {
				90 = {
					trigger_event = {
						id = balkan_crisis.5
						days = { 30 60 }
					}
				}
				10 = {
					trigger_event = {
						id = balkan_crisis.7
						days = 60
					}
				}
			}
		}
	}
	option = { #The Bashi-bazouk can't be trusted, reinforce the local garrisons
		name = balkan_crisis.4.b
		trigger = {
			any_scope_character = {
				is_ruler = no
				is_busy = no
				has_role = general
			}
		}
		random_scope_character = {
			limit = {
				is_ruler = no
				is_busy = no
				has_role = general
			}
			set_character_busy = yes
			set_variable = quelling_uprising_var
		}
		add_modifier = {
			name = quelling_uprising_modifier
		}
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = cu:albanian
				}
			}
			add_radicals_in_state = {
				value = small_radicals
				culture = cu:albanian
			}
		}
		scope:crisis_state.state_region = {
			add_devastation = 15
		}
		scope:crisis_state = {
			add_radicals_in_state = {
				value = very_large_radicals
				culture = scope:crisis_culture
			}
		}
		hidden_effect = {
			if = {
				limit = {
					any_scope_character = {
						has_variable = quelling_uprising_var
						OR = {
							has_trait = bigoted
							has_trait = cruel
							has_trait = pillager
							has_trait = psychological_affliction
							has_trait = war_criminal
						}
					}
				}
				random_list = {
					65 = {
						trigger_event = {
							id = balkan_crisis.5
							days = { 30 60 }
						}
					}
					35 = {
						trigger_event = {
							id = balkan_crisis.7
							days = 60
						}
					}
				}
			}
			else = {
				random_list = {
					35 = {
						trigger_event = {
							id = balkan_crisis.5
							days = { 30 60 }
						}
					}
					65 = {
						trigger_event = {
							id = balkan_crisis.7
							days = 60
						}
					}
				}
			}
		}
	}
}

	type = country_event
	placement = scope:crisis_state
	title = balkan_crisis.5.t
	desc = balkan_crisis.5.d
	flavor =  balkan_crisis.5.f

		video = "unspecific_devastation"
	}

		set_variable = massacre_happened
		random_scope_state = {
			limit = { has_variable = crisis_state_var }
			save_scope_as = crisis_state
		}
		if = {
			limit = { has_variable = greek_uprising }
			cu:greek = { save_scope_as = crisis_culture }
			set_variable = greek_massacre
		}
		if = {
			limit = { has_variable = serb_uprising }
			cu:serb = { save_scope_as = crisis_culture }
			set_variable = serb_massacre
		}
		if = {
			limit = { has_variable = croat_uprising }
			cu:croat = { save_scope_as = crisis_culture }
			set_variable = croat_massacre
		}
		if = {
			limit = { has_variable = bulgarian_uprising }
			cu:bulgarian = { save_scope_as = crisis_culture }
			set_variable = bulgarian_massacre
		}
	}
	trigger = {
		#triggered by random roll on prior event
	}
	option = { #This is a disaster!
		name = balkan_crisis.5.a
		default_option = yes
		change_infamy = 10
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = scope:crisis_culture
				}
			}
			state_region = {
				add_devastation = 10
			}
			kill_population_percent_in_state = {
				percent = 0.02
				culture = scope:crisis_culture
			}
			add_radicals_in_state = {
				value = large_radicals
				culture = scope:crisis_culture
			}
		}
		
		custom_tooltip = {
			text = balkan_crisis_progress_increase_20_tt
			change_global_variable = { name = balkan_crisis_main_var add = 20 }
		}
		
		hidden_effect = {
			trigger_event = {
				id = balkan_crisis.6
				days = 7
			}
			every_country = {
				limit = {
					country_is_in_europe = yes
					any_primary_culture = {
						OR = {
							has_discrimination_trait = western_european_heritage
							has_discrimination_trait = eastern_european_heritage
							has_discrimination_trait = south_european_heritage
						}
					}
				}
				trigger_event = {
					id = balkan_crisis.8
					days = 7
				}
			}
		}
	}
}
