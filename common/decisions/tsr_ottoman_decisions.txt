return_constantinople = {
	is_shown = {
		exists = c:TUR
		OR = {
			has_variable = cyprus_convention
			has_variable = crete_convention
			has_variable = crimea_convention
			has_variable = algiers_convention
		}
		NOT = { has_variable = constantinople_unlawful_territory }
		NOT = { c:TUR ?= this }
		s:STATE_EASTERN_THRACE = {
			any_scope_state = {
				owner = root
			}
		}
	}

	possible = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		s:STATE_EASTERN_THRACE = {
			any_scope_state = {
				owner = root
			}
		}
	}

	when_taken = {
		random_scope_state = {
			limit = { state_region = s:STATE_EASTERN_THRACE }
			set_state_owner = c:TUR
		}
		c:TUR = {
			if = {
				limit = { 
					s:STATE_EASTERN_THRACE = {
						any_scope_state = {
							owner = root
						}
					}
				}
				random_scope_state = {
					limit = { state_region = s:STATE_EASTERN_THRACE }
					set_state_type = incorporated
				}
				set_capital = STATE_EASTERN_THRACE
			}
		}
		change_relations = {
			country = c:TUR
			value = 100
		}
		hidden_effect = {
			every_country = {
				limit = {
					has_claim = s:STATE_EASTERN_THRACE
				}
				s:STATE_EASTERN_THRACE = {
					remove_claim = prev
				}
			}
		}
	}

	ai_chance = {
		base = 0
		modifier = {
			trigger = {
				has_strategy = ai_strategy_maintain_power_balance 
			}
			add = 100
		}
	}
}
