					}

			text = middle_east_crisis_tt

reject_ottoman_caliphate_button = { #Dismiss Caliphatist Ambitions
	name = "reject_ottoman_caliphate_button"
	desc = "reject_ottoman_caliphate_button_desc"

	visible = {
		has_variable = pursue_caliphate_var
		NOT = { has_variable = reject_ottoman_caliphate }
	}

	effect = {
		custom_tooltip = {
			text = dismiss_ottoman_caliphate_tt
			remove_variable = pursue_caliphate_var
			set_variable = reject_ottoman_caliphate
		}
		every_interest_group = {
			limit = {
				has_modifier = pursuing_the_caliphate
			}
			remove_modifier = pursuing_the_caliphate
			add_modifier = {
				name = rejected_the_caliphate
				months = long_modifier_time
				is_decaying = yes
			}
		}
		every_state = {
			limit = {
				region = sr:region_arabic
				has_claim_by = ROOT
			}
			state_region = {
				remove_claim = ROOT
			}
		}
		change_infamy = -10
	}

	ai_chance = {
		base = 0
		modifier = {
			trigger = {
				any_interest_group = {
					is_in_government = yes
					leader = {
						has_ideology = ideology:ideology_pacifist
					}
				}
			}
			add = 20	
		}
	}
}
