	type = country_event
	placement = root
	title = ottoman_flavor.8.t
	desc = ottoman_flavor.8.d
	flavor = ottoman_flavor.8.f

	duration = 3

	event_image = {
		video = "votp_monarch_holding_court"
	}
	
	minor_left_icon = scope:mehmed_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	immediate = {
		random_scope_character = {
			limit = {
				has_template = tur_mehmed_iv_character_template
			}
			save_scope_as = mehmed_scope
		}
		if = {
			limit = {
				ruler = {
					NOT = { has_template = tur_mehmed_iv_character_template }
				}
			}
			ruler = {
				exile_character_with_martyrdom = yes
				save_scope_as = old_ruler_scope
			}
			if = {
				limit = {
					any_scope_character = {
						is_heir = yes
					}
				}
				heir ?= {
					kill_character = yes
				}
			}
			if = {
				limit = {
					any_scope_character = {
						has_variable = prime_minister_trigger
					}
				}
				random_scope_character = {
					limit = { has_variable = prime_minister_trigger }
					remove_variable = prime_minister_trigger
					kill_character = yes
				}
			}
			scope:mehmed_scope = {
				set_character_as_ruler = yes
				remove_character_role = agitator
			}
		}
	}

	option = {
		name = ottoman_flavor.8.a
		default_option = yes
		if = {
			limit = {
				ruler = {
					NOT = { has_template = tur_mehmed_iv_character_template }
				}
			}
			custom_tooltip = {
				text = exile_old_ruler_tt
			}
		}
		ruler = {
			add_modifier = {
				name = the_restorer_modifier
			}
			hidden_effect = {
				if = {
					limit = { has_modifier = mehmed_returns }
					remove_modifier = mehmed_returns
				}
			}
		}
		if = {
			limit = {
				NOR = { 
					has_law = law_type:law_interventionism
					has_law = law_type:law_laissez_faire
				}
			}
			activate_law = law_type:law_laissez_faire
		}
		if = {
			limit = {
				NOR = { 
					has_law = law_type:law_per_capita_based_taxation
					has_law = law_type:law_proportional_taxation
				}
			}
			activate_law = law_type:law_per_capita_based_taxation
		}
		if = {
			limit = {
				NOR = { 
					has_law = law_type:law_freedom_of_conscience
					has_law = law_type:law_shared_roots
				}
			}
			activate_law = law_type:law_freedom_of_conscience
		}
		if = {
			limit = {
				NOT = { 
					has_law = law_type:law_slavery_banned
				}
			}
			activate_law = law_type:law_slavery_banned
		}
		if = {
			limit = {
				NOT = { 
					has_law = law_type:law_appointed_bureaucrats
				}
			}
			activate_law = law_type:law_appointed_bureaucrats
		}
		hidden_effect = {
			every_interest_group = {
				limit = {
					OR = {
						has_ideology = ideology:ideology_westernizer
						leader = {
							has_ideology = ideology:ideology_westernist
						}
					}
				}
				add_modifier = {
					name = westernist_takeover_positive
					months = long_modifier_time
				}
			}
			ig:ig_intelligentsia = {
				add_ruling_interest_group = yes
			}
			every_interest_group = {
				limit = {
					NOR = {
						has_ideology = ideology:ideology_westernizer
						has_ideology = ideology:ideology_pluralist
						has_ideology = ideology:ideology_populist
						leader = {
							has_ideology = ideology:ideology_westernist
						}
					}
				}
				add_modifier = {
					name = westernist_takeover_negative
					months = long_modifier_time
				}

					add_radicals = {
						value = large_radicals

					}
				}

					limit = {
						is_in_government = yes
					}
					remove_ruling_interest_group = yes
				}
			}

				limit = {
