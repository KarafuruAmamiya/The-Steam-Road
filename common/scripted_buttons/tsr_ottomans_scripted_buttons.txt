fractured_porte_caliphatist_button = {
	name = "caliphatist_button"
	desc = "caliphatist_button_desc"

	visible = {
		owner = { 
			NOT = { has_variable = tur_fractured_porte_freeze } 
		}
	}

	possible = {
		trigger_if = {
			limit = {
				has_variable = has_selected_caliphatist_once_var
			}
			custom_tooltip = {
				text = cannot_choose_caliphatist_twice_tt
				always = no
			}
		}
		any_interest_group = {
			is_in_government = yes
			leader = {
				has_ideology = ideology:ideology_caliphatist
			}
		}
		custom_tooltip = {
			text = not_already_supporting_caliphatist
			NOT = { has_variable = supporting_caliphatist_var }
		}
	}
	
	selected = {
		has_variable = supporting_caliphatist_var
	}

	effect = {
		set_variable = supporting_caliphatist_var
		set_variable = has_selected_caliphatist_once_var
		remove_variable = supporting_rumelist_var
		remove_variable = supporting_metropolitan_var
		custom_tooltip = {
			text = caliphatist_progress_increase_10_tt
			change_variable = { name = caliphatist_progress add = 10 }
		}
	}
}

je_turkish_revolution_button_a = {
	name = "je_turkish_revolution_button_a"
	desc = "je_turkish_revolution_button_a_desc"
	
	ai_chance = {
		base = 100
	}

	possible = {
		scope:journal_entry = {
				not = {
				custom_tooltip = {
					text = turkish_revolution_on_cooldown_tt
					has_variable = je_turkish_revolution_decision_cooldown
				}
			}
		}
		OR = {
			custom_tooltip = {
				text = nationalist_ig_low_clout
				any_interest_group = {
					filter = {
						OR = {
							has_ideology = ideology:ideology_nationalist
							leader = {
								OR = {
									has_ideology = ideology:ideology_turkish_nationalist
									has_ideology = ideology:ideology_radical
								}
							}
						}
					}
					count = all
					ig_clout <= 0.1
				}
			}
			custom_tooltip = {
				text = no_nationalist_ig
				NOT = {
					any_interest_group = {
						OR = {
							has_ideology = ideology:ideology_nationalist
							leader = {
								OR = {
									has_ideology = ideology:ideology_turkish_nationalist
									has_ideology = ideology:ideology_radical
								}
							}
						}
					}
				}
			}
		}
	}

	effect = {
        custom_tooltip = cooldown_tur_three_months_tt
		scope:journal_entry = {
			set_variable = { name = je_turkish_revolution_decision_cooldown months = 3 }
		}	
        add_radicals = {
            value = small_radicals
            culture = cu:turkish
        }
        custom_tooltip = revolution_progress_subtract_10
        change_variable = {
            name = turkish_revolution_progress_var
            add = -10
        }
	}
}
