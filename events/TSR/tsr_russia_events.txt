# Moscow Uprising
russian_events.1 = {
	type = country_event
	placement = ROOT

	title = russian_events.1.t
	desc = russian_events.1.d
	flavor = russian_events.1.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	immediate = {
		if = {
			limit = {
				NOT = { has_technology_researched = nationalism }
			}
			add_technology_researched = nationalism
		}
		overlord = {
			save_scope_as = overlord_scope
			set_variable = russian_oppressor
		}
		remove_primary_culture = cu:swedish
		add_primary_culture = cu:russian
		set_state_religion = rel:orthodox
		activate_law = law_type:law_parliamentary_republic
		activate_law = law_type:law_wealth_voting
		if = {
			limit = {
				has_law = law_type:law_hereditary_bureaucrats
			}
			activate_law = law_type:law_appointed_bureaucrats
		}
		create_character = { #WIP
			culture = cu:russian
			interest_group = ig_intelligentsia
			ideology = ideology_radical
			ruler = yes
			traits = {
				ambitious expert_political_operator
			}
		}
	}

	option = {
		name = russian_events.1.a
		default_option = yes
		
		show_as_tooltip = {
			remove_primary_culture = cu:swedish
			add_primary_culture = cu:russian
			set_state_religion = rel:orthodox
		}
		
		custom_tooltip = {
			text = kill_swedes_tt
			every_scope_character = {
				limit = {
					NOT = { culture = cu:russian }
				}
				kill_character = yes
			}
		}
		
		add_journal_entry = { type = je_russian_independence_war }
		
		hidden_effect = {
			every_state = {
				limit = {
					state_region = {
						is_homeland = cu:russian
					}
				}
				state_region = { add_claim = root }
			}
			set_immune_to_revolutions = yes
			scope:overlord_scope = {
				trigger_event = { id = russian_events.2 popup = yes }
				change_relations = {
					country = root
					value = -100
				}
				hidden_effect = {
					set_secret_goal = {
						country = root
						secret_goal = conquer
					}
				}
			}
			every_country = {
				limit = {
					any_primary_culture = {
						has_discrimination_trait = eastern_european_heritage
					}
				}
				trigger_event = { id = russian_events.3 }
			}
		}
	}
}

# Manifesto for the Russian Nation
russian_events.3 = {
	type = country_event
	placement = scope:moscow_state

	title = russian_events.3.t
	desc = russian_events.3.d
	flavor = russian_events.3.f

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	trigger = {
		#Fired by JE
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_technology_researched = nationalism }
			}
			add_technology_researched = nationalism
		}
		c:MSC = {
			capital = {
				save_scope_as = moscow_state
			}
		}
		random_country = {
			limit = { has_variable = russian_oppressor }
			save_scope_as = oppressor_scope
		}
		random_primary_culture = {
			save_scope_as = primary_culture_scope
		}
	}

	option = {
		name = russian_events.3.a #We must fight for our freedom
		default_option = yes
		trigger = {
			is_subject = yes
		}
		if = {
			limit = {
				any_state = {
					is_homeland_of_country_cultures = root
					NOT = { has_claim_by = root }
				}
			}
			custom_tooltip = {
				text = gain_claim_on_homeland_tt
				every_state = {
					limit = {
						is_homeland_of_country_cultures = root
						NOT = { has_claim_by = root }
						owner = { NOT = { is_subject_of = root } }
						owner = { NOT = { root = this } }
					}
					state_region = {
						add_claim = root
					}
				}
			}
		}
		overlord = {
			change_relations = {
				country = root
				value = -100
			}
		}
		every_country = {
			limit = {
				NOT = { root = this }
				NOT = { overlord = this }
				NOT = { is_subject_of = root }
				#NOT = { any_primary_culture = { is_primary_culture_of = root } }
				any_scope_state = {
					is_homeland_of_country_cultures = root
					NOT = { has_claim_by = root }
				}
			}
			change_relations = {
				country = root
				value = -100
			}
		}
	}
	
	option = {
		name = russian_events.3.b #We must reclaim our homeland
		trigger = {
			is_subject = no
		}
		if = {
			limit = {
				any_state = {
					is_homeland_of_country_cultures = root
					NOT = { has_claim_by = root }
				}
			}
			custom_tooltip = {
				text = gain_claim_on_homeland_tt
				every_state = {
					limit = {
						is_homeland_of_country_cultures = root
						NOT = { has_claim_by = root }
						owner = { NOT = { is_subject_of = root } }
						owner = { NOT = { root = this } }
					}
					state_region = {
						add_claim = root
					}
				}
			}
		}
		every_country = {
			limit = {
				NOT = { root = this }
				NOT = { is_subject_of = root }
				#NOT = { any_primary_culture = { is_primary_culture_of = root } }
				any_scope_state = {
					is_homeland_of_country_cultures = root
					NOT = { has_claim_by = root }
				}
			}
			change_relations = {
				country = root
				value = -100
			}
		}
	}
	
	option = {
		name = russian_events.3.b #We cannot alienate our neighbors/overlord
		add_radicals = {
			value = medium_radicals
			culture = scope:primary_culture_scope
		}
		ai_chance = {
			base = 0
		}
	}
}

# End of the Moscow Governorate (victory)
russian_events.4 = {
	type = country_event
	placement = scope:moscow_state

	title = russian_events.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = russian_events.4.d
				trigger = {
					c:MSC ?= this
				}
			}
			triggered_desc = {
				desc = russian_events.4.d2
				trigger = { 
					OR = {
						c:SWE ?= this 
						c:SCA ?= this 
					}
				}
			}
		}
	}
	flavor = russian_events.4.f

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 3

	trigger = {
		#Fired by JE
	}

	immediate = {
		random_state = {
			limit = {
				state_region = s:STATE_MOSCOW
			}
			save_scope_as = moscow_state
		}
	}

	option = {
		name = russian_events.4.a
		default_option = yes
		trigger = { c:MSC ?= this }
		add_modifier = {
			name = won_russian_independence_war
			months = normal_modifier_time
		}
		every_state = {
			limit = {
				OR = {
					region = sr:region_central_russia
					region = sr:region_volga
					region = sr:region_barents
					state_region = {
						is_homeland = cu:russian
					}
				}
				NOT = { has_claim_by = root }
				NOT = { owner = root }
				NOT = { owner = { is_subject_of = root } }
			}
			state_region = {
				add_claim = root
			}
		}
		hidden_effect = {
			if = {
				limit = { has_modifier = russian_independence_war }
				remove_modifier = russian_independence_war
			}
			if = {
				limit = { has_modifier = russian_army_modifier }
				remove_modifier = russian_army_modifier
			}
			if = {
				limit = { has_modifier = modifier_russian_hiring_bonus }
				remove_modifier = modifier_russian_hiring_bonus
			}
		}
	}
	
	option = {
		name = russian_events.4.b
		trigger = { 
			OR = {
				c:SWE ?= this 
				c:SCA ?= this 
			}
		}
		add_modifier = {
			name = lost_russian_independence_war
			months = normal_modifier_time
		}
	}
}
