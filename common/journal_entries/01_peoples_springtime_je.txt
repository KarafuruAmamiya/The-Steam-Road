je_springtime_of_the_peoples = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_crises

	is_shown_when_inactive = {
		total_population > 300000 #no springtime in small countries
		has_technology_researched = egalitarianism
		year <= 1875
		capital = {
			OR = {
				region = sr:region_england
				region = sr:region_gaelic
				region = sr:region_iberia
				region = sr:region_france
				region = sr:region_rhine
				region = sr:region_low_countries
				region = sr:region_italy
				region = sr:region_balkans
				region = sr:region_greece
				region = sr:region_north_germany
				region = sr:region_south_germany
				region = sr:region_baltic
				region = sr:region_scandinavia
				region = sr:region_poland
				region = sr:region_dnieper
				region = sr:region_central_russia
				#region = sr:region_anatolia
				#region = sr:region_persia
			}
		}
		NOR = {
			coa_republic_trigger = yes
			has_law = law_type:law_census_voting
			has_law = law_type:law_universal_suffrage	
		}
	}

	possible = {
		any_interest_group = {
			OR = {
				ig_approval < -10 # ANGERY
				is_insurrectionary = yes
			}
			leader = { has_ideology = ideology:ideology_radical }
		}
		is_revolutionary = no
	}

	immediate = {
		trigger_event = { id = 1848.4 }
		hidden_effect = {
			if = {
				limit = {
					is_revolutionary = no
					}
				save_scope_as = originator
				every_country = {
					limit = {
						OR = {
							capital = {
								OR = {
									region = sr:region_england
									region = sr:region_gaelic
									region = sr:region_iberia
									region = sr:region_france
									region = sr:region_rhine
									region = sr:region_low_countries
									region = sr:region_italy
									region = sr:region_balkans
									region = sr:region_greece
									region = sr:region_north_germany
									region = sr:region_south_germany
									region = sr:region_baltic
									region = sr:region_scandinavia
									region = sr:region_poland
									region = sr:region_dnieper
									region = sr:region_central_russia
									#region = sr:region_anatolia
									#region = sr:region_persia
								}
							} 
							has_interest_marker_in_region = scope:originator.capital.region
						}
						NOT = { this = root }
					}
					post_notification = springtime_of_the_people_je_opened
				}
			}
		}
	}

	complete = {
		is_at_war = no
		has_revolution = no
		is_revolutionary = no
		has_insurrectionary_interest_groups = no
		NOT = { has_law = law_type:law_monarchy }
		OR = {
			has_law = law_type:law_census_voting
			has_law = law_type:law_universal_suffrage
		}	
	}

	on_complete = {
		set_variable = {
			name = completed_peoples_stringtime
		}
		set_variable = {
			name = completed_peoples_stringtime_recent
			months = short_modifier_time
		}
		trigger_event = { id = peoples_springtime.7 }
	}

	on_monthly_pulse = {
		events = {
			1848.5
		}
		random_events = {
			100 = 0
			10 = peoples_springtime.3
			10 = peoples_springtime.4
			10 = peoples_springtime.5
		}
	}
	
	modifiers_while_active = {
		1848_modifier
	}

	weight = 10000
	should_be_pinned_by_default = yes

	transferable = no
	can_revolution_inherit = no

	timeout = 3650

	on_timeout = {
		hidden_effect = {
			if = {
				limit = {
					is_revolutionary = no
					}
				save_scope_as = originator
				every_country = {
					limit = {
						OR = {
							capital = {
								OR = {
									region = sr:region_england
									region = sr:region_gaelic
									region = sr:region_iberia
									region = sr:region_france
									region = sr:region_rhine
									region = sr:region_low_countries
									region = sr:region_italy
									region = sr:region_balkans
									region = sr:region_greece
									region = sr:region_north_germany
									region = sr:region_south_germany
									region = sr:region_baltic
									region = sr:region_scandinavia
									region = sr:region_poland
									region = sr:region_dnieper
									region = sr:region_central_russia
									#region = sr:region_anatolia
									#region = sr:region_persia
								}
							} 
							has_interest_marker_in_region = scope:originator.capital.region
						}
						NOT = { this = root }
					}
					post_notification = springtime_of_the_people_je_over
				}
				trigger_event = { id = peoples_springtime.8 }
			}
		}
	}
}
