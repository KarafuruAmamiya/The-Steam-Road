on_weekly_pulse = {
	effect = {
		every_scope_building = {
			limit = {
                is_building_type = building_railway
            }
			if = {
                limit = {
                    root = { is_player = no }
                    is_subsidized = no
                }
                set_subsidized = yes
            }
		}
	}
}

	events = {
		arabia_flavor.1

		arabia_flavor.3
		arabia_flavor.4
		opium_events.1
		persian_flavor.13
		persian_flavor.3
	}
	effect = {
		clear_global_variable_list = council_republics
		every_country = {
			limit = { 
				has_law = law_type:law_council_republic
				country_rank > rank_value:minor_power 
			}
			add_to_global_variable_list = {
				name = council_republics
				target = THIS
			}
		}
		if = {
			limit = {
				c:HDJ ?= this
				NOT = {
					any_scope_state = {
						state_region = s:STATE_HEDJAZ
					}
				}
				has_modifier = holy_city_of_mecca
			}
			remove_modifier = holy_city_of_mecca
		}
		if = {
			limit = {
				any_scope_state = {
					state_region = {
						is_homeland = cu:ruthenian
					}
				}
				OR = {
					NOR = { 
						country_has_primary_culture = cu:ruthenian
						country_has_primary_culture = cu:cossack
					}
					c:LIT ?= this
					c:GAL ?= this
				}
				NOT = {
					any_country = {
						NOR = {
							c:LIT ?= this
							c:GAL ?= this
						}
						country_has_primary_culture = cu:ruthenian
					}
				}
				NOT = { has_variable = ruthenian_culture_change }
			}
			trigger_event = { id = ruthenian_events.5 }
		}
	}
}

tsr_diplomatic_play_started = {
	effect = {
		if = {
			limit = {
				is_diplomatic_play_type = dp_independence
			}
			random_country = { 
				limit = {
					has_journal_entry = je_arab_nationalism_subject
					has_variable = independence_plot_leader
					is_subject_of = c:AKN
				}
				trigger_event = { id = arab_wars.8 days = 0 } 
			}
		}
		if = {
			limit = {
				is_diplomatic_play_type = dp_independence
			}
			random_country = { 
				limit = {
					has_journal_entry = je_arab_nationalism_subject
					c:EGY ?= ROOT
					is_subject_of = c:TUR
				}
				trigger_event = { id = arab_wars.17 days = 0 } 
			}
		}
		if = {
			limit = {
				is_diplomatic_play_type = dp_opium_wars
				scope:initiator = {
					NOT = { has_variable = FOW_decision_taken }
				}
			}
			scope:initiator = {
				trigger_event = { id = opium_wars.9 days = 0 }
			}
		}
		if = {
			limit = {
				scope:initiator = {
					has_variable = balkan_war_attacker
					has_journal_entry = je_war_of_the_balkan_coalition_attacker
				}
				scope:target = {
					has_journal_entry = je_war_of_the_balkan_coalition_defender
				}
			}
			scope:initiator = {
				trigger_event = { id = balkan_crisis_2.2 days = 0 }
			}
		}
	}
}
