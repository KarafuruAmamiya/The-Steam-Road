je_akkovin_crisis_button = {
	name = "je_akkovin_crisis_button"
	desc = "je_akkovin_crisis_button_desc"

	visible = { NOT = { has_variable = erzurum_edict } }

	possible = {
		OR = {
			ig:ig_intelligentsia = {
				is_in_government = yes
			}
			any_scope_character = {
				is_ruler = yes
				interest_group = {
					is_interest_group_type = ig_intelligentsia
				}
			}
		}
	}

	effect = {
		set_variable = erzurum_edict
		add_radicals = {
			value = small_radicals
		}
		hidden_effect = { trigger_event = { id = akkovin_crisis.5 days = 90 popup = yes } }
	}
	
	ai_chance = {
		base = 100
	}
}
