scandinavian_union_denmark_button = {
	name = "scandinavian_union_denmark_button"
	desc = "scandinavian_union_denmark_button_desc"

	visible = { 
		exists = c:DEN
	}
	
	ai_chance = {
		base = 100
	}

	possible = {
		custom_tooltip = {
			text = denmark_point_10
			global_var:denmark_point >= 10
		}
	}

	effect = {
		set_variable = denmark_appeased
		if = {
			limit = {
				c:DEN ?= {
					is_subject_of = root
					NOR = {
						is_subject_type = subject_type_dependency
						is_subject_type = subject_type_real_union
					}
				}
			}
			if = {
				limit = {
					c:DEN ?= {
						is_subject_type = subject_type_personal_union
					}
				}
				c:DEN ?= {
					change_subject_type = subject_type_real_union
				}
			}
			else = {
				c:DEN ?= {
					change_subject_type = subject_type_dependency
				}
			}
		}
	}
}
