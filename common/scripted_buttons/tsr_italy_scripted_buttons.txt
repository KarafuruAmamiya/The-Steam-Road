je_latin_republic_transfer_state_button = {
	name = "je_latin_republic_transfer_state_button"
	desc = "je_latin_republic_transfer_state_button_desc"

	visible = { always = yes }
	
	ai_chance = {
		base = 100
	}

	possible = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		OR = {
			sr:region_greece = {
				any_scope_state = { owner = root }
			}
			sr:region_balkans = {
				any_scope_state = { owner = root }
			}
			sr:region_anatolia = {
				any_scope_state = { owner = root }
			}
		}
		trigger_if = {
			limit = { exists = c:LAE }
			c:LAE ?= { is_subject_of = root }
		}
		trigger_else_if = {
			limit = { exists = c:GRE }
			c:GRE ?= { is_subject_of = root }
		}
		trigger_else = {
			any_subject_or_below = {
				OR = {
					capital.region = sr:region_greece
					capital.region = sr:region_balkans
					capital.region = sr:region_anatolia
				}
			}
		}
	}

	effect = {
		every_scope_state = {
			limit = {
				owner = ROOT
				OR = {
					region = sr:region_greece
					region = sr:region_balkans
					region = sr:region_anatolia
				}
			}
			if = {
				limit = { 
					exists = c:LAE 
					c:LAE ?= { is_subject_of = root }
				}
				set_state_owner = c:LAE
				set_state_type = unincorporated
			}
			else_if = {
				limit = { 
					exists = c:GRE 
					c:GRE ?= { is_subject_of = root }
				}
				set_state_owner = c:GRE
				set_state_type = unincorporated
			}
			else = {
				random_subject_or_below = {
					limit = {
						OR = {
							capital.region = sr:region_greece
							capital.region = sr:region_balkans
							capital.region = sr:region_anatolia
						}
					}
					save_temporary_scope_as = balkan_subject_scope
				}
				set_state_owner = scope:balkan_subject_scope
				set_state_type = unincorporated
			}
		}
	}
}

je_latin_republic_form_latin_rep_button = {
	name = "je_latin_republic_form_latin_rep_button"
	desc = "je_latin_republic_form_latin_rep_button_desc"

	visible = { NOT = { has_variable = formed_latin_republic } }
	
	ai_chance = {
		base = 100
	}

	possible = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		OR = {
			AND = {
				custom_tooltip = {
					text = own_all_greece_states
					NOT = {
						sr:region_greece = {
							any_scope_state = {
								NOT = { owner = ROOT }
							}
						}
					}
				}
				calc_true_if = {
					amount >= 4
					owns_entire_state_region = STATE_HUDAVENDIGAR
					owns_entire_state_region = STATE_EASTERN_THRACE
					owns_entire_state_region = STATE_AYDIN
					owns_entire_state_region = STATE_KARASI
					owns_entire_state_region = STATE_EAST_AEGEAN_ISLANDS
					owns_entire_state_region = STATE_NORTHERN_THRACE
					owns_entire_state_region = STATE_SKOPIA
				}
			}
			any_subject_or_below = {
				custom_tooltip = {
					text = own_all_greece_states
					NOT = {
						sr:region_greece = {
							any_scope_state = {
								NOT = { owner = { is_subject_of = root } }
							}
						}
					}
				}
				calc_true_if = {
					amount >= 4

					owns_entire_state_region = STATE_EASTERN_THRACE
					owns_entire_state_region = STATE_AYDIN
					owns_entire_state_region = STATE_KARASI
					owns_entire_state_region = STATE_EAST_AEGEAN_ISLANDS
					owns_entire_state_region = STATE_NORTHERN_THRACE
					owns_entire_state_region = STATE_SKOPIA
				}
			}
		}
	}

	effect = {
		set_variable = formed_latin_republic
		if = {
			limit = {
				any_subject_or_below = {
					custom_tooltip = {
						text = own_all_greece_states
						NOT = {
							sr:region_greece = {
								any_scope_state = {
									NOT = { owner = { is_subject_of = root } }
								}
							}
						}
					}
				}
				calc_true_if = {
					amount >= 4
					owns_entire_state_region = STATE_HUDAVENDIGAR
					owns_entire_state_region = STATE_EASTERN_THRACE
					owns_entire_state_region = STATE_AYDIN
					owns_entire_state_region = STATE_KARASI
					owns_entire_state_region = STATE_EAST_AEGEAN_ISLANDS
					owns_entire_state_region = STATE_NORTHERN_THRACE
					owns_entire_state_region = STATE_SKOPIA
				}
			}
			trigger_event = { id = latin_republic.4 popup = yes }
		}
		else = {
			trigger_event = { id = latin_republic.5 popup = yes }
		}
	}
}
