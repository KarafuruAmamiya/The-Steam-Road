je_turkish_revolution = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	group = je_group_tsr_content
	
	can_revolution_inherit = no
	
	modifiers_while_active = {
		modifier_tur_nationalism_2
	}

	immediate = {
		set_global_variable = turkish_revolution_happened
		set_variable = {
			name = turkish_revolution_progress_var
			value = 0
		}
	}

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		has_law = law_type:law_monarchy
		NOR = {
			has_law = law_type:law_census_voting
			has_law = law_type:law_universal_suffrage
			has_law = law_type:law_presidential_republic
			has_law = law_type:law_parliamentary_republic
		}
		any_revolution_ongoing = no
	}

	on_complete = {
		trigger_event = { id = turkish_nationalism.5 popup = yes }
	}

	fail = {
		OR = {
			has_revolution_over_law_type = law_type:law_census_voting
			has_revolution_over_law_type = law_type:law_universal_suffrage
			has_revolution_over_law_type = law_type:law_presidential_republic
			has_revolution_over_law_type = law_type:law_parliamentary_republic
		}
		any_country = {
			is_revolutionary = yes
			country_definition = root.country_definition
			OR = {
				has_law = law_type:law_census_voting
				has_law = law_type:law_universal_suffrage
				has_law = law_type:law_presidential_republic
				has_law = law_type:law_parliamentary_republic
			}					
		}
	}

	on_fail = {
		trigger_event = { id = turkish_nationalism.7 popup = yes }
	}
	
	invalid = {
		OR = {
			NOT = { has_law = law_type:law_monarchy }
			has_law = law_type:law_census_voting
			has_law = law_type:law_universal_suffrage
		}
	}
	
	on_invalid = {
		trigger_event = { id = turkish_nationalism.6 popup = yes }
		hidden_effect = {
			every_interest_group = {
				limit = {
					has_modifier = westernist_takeover_positive
				}
				remove_modifier = westernist_takeover_positive
			}
			every_interest_group = {
				limit = {
					has_modifier = westernist_takeover_negative
				}
				remove_modifier = westernist_takeover_negative
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_law = law_type:law_monarchy
					NOR = {
						has_law = law_type:law_census_voting
						has_law = law_type:law_universal_suffrage
						has_law = law_type:law_presidential_republic
						has_law = law_type:law_parliamentary_republic
					}
					any_revolution_ongoing = no
				}
				change_variable = {
					name = turkish_revolution_progress_var
					add = 1
				}
			}
			else = {
				change_variable = {
					name = turkish_revolution_progress_var
					subtract = 1
					min = 0
				}
			}
		}
	}

	should_be_pinned_by_default = yes

	weight = 10000
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = turkish_revolution_not_progressing
				trigger = {
					has_law = law_type:law_monarchy
					NOR = {
						has_law = law_type:law_census_voting
						has_law = law_type:law_universal_suffrage
						has_law = law_type:law_presidential_republic
						has_law = law_type:law_parliamentary_republic
					}
					any_revolution_ongoing = no
				}
			}
			triggered_desc = {
				desc = turkish_revolution_progressing
				trigger = {
					any_revolution_ongoing = yes
				}
			}
			triggered_desc = {
				desc = turkish_revolution_complete
				trigger = {
					NOT = {
						has_law = law_type:law_monarchy
						NOR = {
							has_law = law_type:law_census_voting
							has_law = law_type:law_universal_suffrage
							has_law = law_type:law_presidential_republic
							has_law = law_type:law_parliamentary_republic
						}
					}
					any_revolution_ongoing = no
				}
			}
		}
	}

	current_value = {
		value = var:turkish_revolution_progress_var
	}

	goal_add_value = {
		add = 120
	}

	progressbar = yes
}
