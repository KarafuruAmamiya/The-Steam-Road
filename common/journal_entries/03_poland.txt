je_poland_lithuania = {
    icon = "gfx/interface/icons/event_icons/event_flags/event_plc_flag.dds"

	group = je_group_foreign_affairs

    is_shown_when_inactive = {
       c:PLC ?= this
    }

    possible = {
        has_technology_researched = pan-nationalism
        custom_tooltip = {
            text = odrodzenie_complete_tt
           has_variable = odrodzenie_complete
        }
    }

    immediate = {
        trigger_event = { id = poland_events.6 popup = yes }
    }

    complete = {
        owns_entire_state_region = STATE_VITEBSK
        owns_entire_state_region = STATE_KIEV
        any_scope_state = {
            state_region = {
                OR = {
                    is_homeland = cu:ukrainian
                    is_homeland = cu:byelorussian
                    is_homeland = cu:ruthenian
                }
            }
            count >=7
        }
    }

    on_complete = {
        set_variable = greater_commonwealth_var
        trigger_event = { id = poland_events.7 popup = yes }
        custom_tooltip = plc_discrimination_tt
        show_as_tooltip = {
			if = {
				limit = {
					any_country = {
						country_has_primary_culture = cu:ukrainian
					}
				}
				add_primary_culture = cu:ukrainian
				add_loyalists = {
					culture = cu:ukrainian
					value = 0.1
				}
			}
			if = {
				limit = {
					any_country = {
						country_has_primary_culture = cu:byelorussian
					}
				}
				add_primary_culture = cu:byelorussian
				add_loyalists = {
					culture = cu:byelorussian
					value = 0.1
				}
			}
			if = {
				limit = {
					NOT = {
						any_country = {
							OR = {
								country_has_primary_culture = cu:ukrainian
								country_has_primary_culture = cu:byelorussian
							}
						}
					}
				}
				add_primary_culture = cu:ruthenian
				add_loyalists = {
					culture = cu:ruthenian
					value = 0.1
				}
			}
        }
        custom_tooltip = plc_discrimination_tt_2
        show_as_tooltip = {
            add_loyalists = {
                culture = cu:polish
                value = 0.1
            }
        }
    }

    should_be_pinned_by_default = yes
}
