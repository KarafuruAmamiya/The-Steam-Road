remove_denmark_point_effect = {
	custom_tooltip = {
		text = remove_denmark_point
		change_global_variable = {
			name = denmark_point
			add = -1
		}
	}
}

add_denmark_point_effect = {
	custom_tooltip = {
		text = add_denmark_point
		change_global_variable = {
			name = denmark_point
			add = 1
		}
	}
}
