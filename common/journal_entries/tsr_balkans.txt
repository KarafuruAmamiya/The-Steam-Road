je_balkan_crisis_tur = {
	icon = "gfx/interface/icons/event_icons/event_revolt.dds"

	group = je_group_tsr_content

	is_shown_when_inactive = {
		NOT = { has_global_variable = balkan_crisis_happened }
		c:TUR ?= this
		has_technology_researched = nationalism
		country_rank >= rank_value:major_power
	}

	possible = {
		hidden_trigger = {
			NOT = { has_global_variable = balkan_crisis_happened }
		}
		OR = {
			custom_tooltip = {
				text = lost_war_tt
				OR = {
					has_variable = recently_lost_war
					AND = {
						has_variable = recently_had_war
						NOT = {
							has_variable = recently_won_war
						}
					}
				}
			}
			in_default = yes
			tax_level >= high
			custom_tooltip = {
				text = turkish_revolution_happened_tt
				has_global_variable = turkish_revolution_happened
			}
		}
		any_scope_state = {
			OR = {
				region = sr:region_greece
				region = sr:region_balkans
			}
			turmoil >= 0.2
		}
	}

	immediate = {
		c:TUR ?= { save_scope_as = tur_scope }
		trigger_event = { id = balkan_crisis.1 popup = yes }
		set_global_variable = balkan_crisis_happened
		set_global_variable = {
			name = balkan_crisis_main_var
			value = 0
		}
		set_variable = {
			name = greek_states_radicals_total_final
			value = 0
		}
		set_variable = {
			name = serb_states_radicals_total_final
			value = 0
		}
		set_variable = {
			name = croat_states_radicals_total_final
			value = 0
		}
		set_variable = {
			name = bulgarian_states_radicals_total_final
			value = 0
		}
		set_balkan_crisis_vars = yes
	}
	
	complete = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		has_technology_researched = pan-nationalism
		AND = {
			OR = {
				has_law = law_type:law_total_separation
				has_law = law_type:law_shared_roots
			}
			OR = {
				has_law = law_type:law_cultural_exclusion
				has_law = law_type:law_multicultural
			}
			OR = {
				has_law = law_type:law_federal
				has_law = law_type:law_confederated
			}
		}
		any_scope_state = {
			filter = {
				OR = {
					region = sr:region_greece
					region = sr:region_balkans
				}
			}
			turmoil < 0.2
			count = all
		}
	}

	fail = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		NAND = {
			trigger_if = {
				limit = { exists = c:MAC }
				c:MAC ?= {
					is_subject_of = root
				}
			}
			trigger_else = {
				country_or_subject_owns_entire_state_region = STATE_WESTERN_THRACE
			}
			trigger_if = {
				limit = { exists = c:SER }
				c:SER ?= {
					is_subject_of = root
				}
			}
			trigger_else = {
				country_or_subject_owns_entire_state_region = STATE_SOUTHERN_SERBIA
				country_or_subject_owns_entire_state_region = STATE_NORTHERN_SERBIA
				country_or_subject_owns_entire_state_region = STATE_BOSNIA
				country_or_subject_owns_entire_state_region = STATE_MONTENEGRO
			}
			trigger_if = {
				limit = { exists = c:CRO }
				c:CRO ?= {
					is_subject_of = root
				}
			}
			trigger_else = {
				country_or_subject_owns_entire_state_region = STATE_CROATIA
				country_or_subject_owns_entire_state_region = STATE_SLAVONIA
			}
			trigger_if = {
				limit = { exists = c:BUL }
				c:BUL ?= {
					is_subject_of = root
				}
			}
			trigger_else = {
				country_or_subject_owns_entire_state_region = STATE_BULGARIA
				country_or_subject_owns_entire_state_region = STATE_WEST_BULGARIA
				country_or_subject_owns_entire_state_region = STATE_DOBRUDJA
				country_or_subject_owns_entire_state_region = STATE_NORTHERN_THRACE
			}
		}
	}
	
	on_complete = {
		set_global_variable = balkan_crisis_complete
		trigger_event = { id = balkan_crisis_2.3 popup = yes }
		clear_balkan_crisis_vars = yes
		remove_global_variable = balkan_crisis_main_var
		remove_global_variable = greek_states_radicals_total_final
		remove_global_variable = serb_states_radicals_total_final
		remove_global_variable = croat_states_radicals_total_final
		remove_global_variable = bulgarian_states_radicals_total_final
		every_country = {
			limit = { has_variable = balkan_war_attacker }
			remove_variable = balkan_war_attacker
		}
	}
	
	on_fail = {
		set_global_variable = balkan_crisis_fail_var
		custom_tooltip = {
			text = balkans_fail_condition_1_tt
			trigger_event = { id = balkan_crisis_2.4 popup = yes }
		}
		custom_tooltip = {
			text = balkans_fail_condition_2_tt
			trigger_event = { id = balkan_crisis_2.5 popup = yes }
		}
		clear_balkan_crisis_vars = yes
		remove_global_variable = balkan_crisis_main_var
		remove_global_variable = greek_states_radicals_total_final
		remove_global_variable = serb_states_radicals_total_final
		remove_global_variable = croat_states_radicals_total_final
		remove_global_variable = bulgarian_states_radicals_total_final
		every_country = {
			limit = { has_variable = balkan_war_attacker }
			remove_variable = balkan_war_attacker
		}
	}
	
	on_weekly_pulse = {
		effect = {
			calculate_balkan_unrest = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			calculate_balkan_unrest = yes
			add_balkan_crisis_var = yes
			
			if = {
				limit = {
					NOT = { has_global_variable = constantinople_conference }
					c:TUR ?= { has_variable = massacre_happened }
					any_country = {
						country_is_in_europe = yes
						country_rank >= rank_value:great_power 
						NOT = { c:TUR ?= this }
						NOT = { is_diplomatic_play_enemy_of = c:TUR }
						count >= 5
					}
				}
				set_global_variable = constantinople_conference
				random_country = {
					limit = {
						country_is_in_europe = yes
						country_rank >= rank_value:great_power 
						NOT = { c:TUR ?= this }
					}
					set_variable = european_gp_1
					create_truce = {
						country = c:TUR
						months = 12
					}
				}
				random_country = {
					limit = {
						country_is_in_europe = yes
						country_rank >= rank_value:great_power 
						NOT = { c:TUR ?= this }
						NOT = { has_variable = european_gp_1 }
					}
					set_variable = european_gp_2
					create_truce = {
						country = c:TUR
						months = 12
					}
				}
				random_country = {
					limit = {
						country_is_in_europe = yes
						country_rank >= rank_value:great_power 
						NOT = { c:TUR ?= this }
						NOT = { has_variable = european_gp_1 }
						NOT = { has_variable = european_gp_2 }
					}
					set_variable = european_gp_3
					create_truce = {
						country = c:TUR
						months = 12
					}
				}
				random_country = {
					limit = {
						country_is_in_europe = yes
						country_rank >= rank_value:great_power 
						NOT = { c:TUR ?= this }
						NOT = { has_variable = european_gp_1 }
						NOT = { has_variable = european_gp_2 }
						NOT = { has_variable = european_gp_3 }
					}
					set_variable = european_gp_4
					create_truce = {
						country = c:TUR
						months = 12
					}
				}
				random_country = {
					limit = {
						country_is_in_europe = yes
						country_rank >= rank_value:great_power 
						NOT = { c:TUR ?= this }
						NOT = { has_variable = european_gp_1 }
						NOT = { has_variable = european_gp_2 }
						NOT = { has_variable = european_gp_3 }
						NOT = { has_variable = european_gp_4 }
					}
					set_variable = european_gp_5
					create_truce = {
						country = c:TUR
						months = 12
					}
				}
				random_country = {
					limit = { has_variable = european_gp_1 }
					trigger_event = { id = balkan_crisis.9 popup = yes }
				}
				random_country = {
					limit = { has_variable = european_gp_2 }
					trigger_event = { id = balkan_crisis.9 popup = yes }
				}
				random_country = {
					limit = { has_variable = european_gp_3 }
					trigger_event = { id = balkan_crisis.9 popup = yes }
				}
				random_country = {
					limit = { has_variable = european_gp_4 }
					trigger_event = { id = balkan_crisis.9 popup = yes }
				}
				random_country = {
					limit = { has_variable = european_gp_5 }
					trigger_event = { id = balkan_crisis.9 popup = yes }
				}
			}
		}
		events = { 
			balkan_crisis.3 #Balkan Uprising
		}
	}
	
	current_value = {
		value = global_var:balkan_crisis_main_var
	}

	goal_add_value = {
		add = 120
	}
	
	invalid = {
		calc_true_if = {
			amount >= 8
			s:STATE_WESTERN_THRACE = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_SOUTHERN_SERBIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_NORTHERN_SERBIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_BOSNIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_MONTENEGRO = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_CROATIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_SLAVONIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_BULGARIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_WEST_BULGARIA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_DOBRUDJA = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
			s:STATE_NORTHERN_THRACE = {
				any_scope_state = {
					owner = { is_subject_of = root }
				}
			}
		}
	}
	
	on_invalid = {
		trigger_event = { id = balkan_crisis_2.4 popup = yes }
		clear_balkan_crisis_vars = yes
		remove_global_variable = balkan_crisis_main_var
		remove_global_variable = greek_states_radicals_total_final
		remove_global_variable = serb_states_radicals_total_final
		remove_global_variable = croat_states_radicals_total_final
		remove_global_variable = bulgarian_states_radicals_total_final
		every_country = {
			limit = { has_variable = balkan_war_attacker }
			remove_variable = balkan_war_attacker
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = balkan_crisis_turbulent
				trigger = {
					global_var:balkan_crisis_main_var >= 100
				}
			}
			triggered_desc = {
				desc = balkan_crisis_volatile
				trigger = {
					global_var:balkan_crisis_main_var >= 50
				}
			}
			triggered_desc = {
				desc = balkan_crisis_uneasy
				trigger = {
					global_var:balkan_crisis_main_var < 50
				}
			}
		}
	}

	progressbar = yes

	should_be_pinned_by_default = yes

	weight = 8000
}
