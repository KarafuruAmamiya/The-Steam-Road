tsr_misc_events.2 = {
	type = country_event
	hidden = yes

	trigger = {
		has_law = law_type:law_governorate
		country_has_voting_franchise = no
		NOT = { has_variable = memento_mori }
		ruler = {
			NOT = { has_variable = memento_mori_immunity }
		}
	}

	immediate = {
		ruler = {
			save_scope_as = memento_mori_leader
		}
		set_variable = {
			name = memento_mori
			years = 5
		}
		scope:memento_mori_leader = {
			if = {
				limit = {
					is_historical = yes
				}
				random_list = {
					6 = {
						kill_character = {
							hidden = yes
						}
						owner = {
							if = {
								limit = { has_variable = primary_culture_ruler }
								overlord = {
									ruler.culture = {
										save_scope_as = overlord_culture
									}
									religion = {
										save_scope_as = overlord_religion
									}
								}
								create_character = {
									ruler = yes
									culture = scope:overlord_culture
									religion = scope:overlord_religion
								}
							}
							ruler = {
								set_variable = {
									name = memento_mori_immunity
									months = 60
								}
							}
						}
					}
					3 = {}
				}
			}
			else = {
				kill_character = {
					hidden = yes
				}
				owner = {
					if = {
						limit = { has_variable = primary_culture_ruler }
						overlord = {
							ruler.culture = {
								save_scope_as = overlord_culture
							}
							religion = {
								save_scope_as = overlord_religion
							}
						}
						create_character = {
							ruler = yes
							culture = scope:overlord_culture
							religion = scope:overlord_religion
						}
					}
					ruler = {
						set_variable = {
							name = memento_mori_immunity
							months = 60
						}
					}
				}
			}
		}
	}
}
