je_russian_nationalism = {
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_tsr_content
	
    immediate = {
		hidden_effect = {
			set_variable = {
				name = russian_unrest_je_var
				value = 0
			}
		}
		set_russian_unrest_vars = yes
    }
	
	on_weekly_pulse = {
		effect = {
			calculate_russian_unrest = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			calculate_russian_unrest = yes
			add_russian_unrest = yes
		}
	}
	
    complete = {
        custom_tooltip = {
			text = russian_unrest_variable_at_full_tt
			root.var:russian_unrest_je_var >= 100
		}
    }
	
    on_complete = {
        trigger_event = { id = russian_events.1 }
    }
	
    fail = {
        
    }

	current_value = {
		value = root.var:russian_unrest_je_var
	}

	goal_add_value = {
		add = 100
	}
	
	invalid = {
		is_subject = no
	}
	
	on_invalid = {
		remove_primary_culture = cu:swedish
		add_primary_culture = cu:russian
		every_scope_character = {
			limit = {
				NOT = { culture = cu:russian }
			}
			kill_character = yes
		}
	}

	progressbar = yes

    weight = 1000
    should_be_pinned_by_default = yes
}

je_moscow_uprising = {
    icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_tsr_content

    complete = {
		is_at_war = no
        OR = {
			NOT = { exists = c:MSC }
			c:MSC = {
                is_subject_of = root 
            }
        }
    }
    on_complete = {
		if = {
			limit = {
				exists = c:MSC
			}
			c:MSC = {
				hidden_effect = {
					set_immune_to_revolutions = no
				}
			}
		}
        trigger_event = { id = russian_events.4 }
    }
    fail = {
		is_at_war = no
        trigger_if = {
            limit = {
                exists = c:MSC
            }
            NOT = {
                c:MSC = {
                    is_subject_of = root
                }
            }
        }
		trigger_else = {
            NOT = { owns_entire_state_region = STATE_MOSCOW }
        }
    }
	on_fail = {
		if = {
			limit = {
				exists = c:MSC
			}
			c:MSC = {
				hidden_effect = {
					set_immune_to_revolutions = no
				}
			}
		}
	}

    weight = 1000
    should_be_pinned_by_default = yes
}
