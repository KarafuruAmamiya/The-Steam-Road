vfm_released_independent_recognized = {
	effect = {
		if = {
			limit = {
				OR = {
					is_country_type = recognized
					is_country_type = colonial
				}
				country_rank >= rank_value:great_power
				scope:target = {
					is_country_type = unrecognized
				}
			}
			scope:target = {
				set_country_type = recognized
			}
		}
	}
}

vfm_barracks_colony_created = {
	effect = {
		#temporary solution giving barracks to colonies to prevent being overrun by uprisings instantly
		if = {
			limit = {
				owner = {
					country_is_in_africa = no
					is_ai = yes
				}
				is_under_colonization = yes
				is_incorporated = no
				OR = {
					state_in_africa = yes
				}
				NOT = {
					has_building = building_barracks
				}
			}
			create_building = {
				building = building_barracks
				level = 1
			}
		}
	}
}

vfm_vice_presidents = {
	effect = {
		if = { #Creates initial vice president if conditions met
			limit = {
				has_law = law_type:law_presidential_republic
				NOT = { has_variable = vice_president }
			}
			random_scope_character = {
				limit = {
					is_ruler = no
					is_heir = no
					NOR = {
						has_role = general
						has_role = admiral
						has_role = agitator
						has_variable = vice_president_trigger
					}
					interest_group = {
						is_marginal = no
						ig_approval > 0
						ig_clout >= 0.1
					}
				}
				set_variable = vice_president_trigger
				save_scope_as = new_vice_president
				owner = {
					set_variable = {
						name = vice_president
						value = PREV
					}
				}
			}
		}
		if = { #New VP
			limit = {
				has_law = law_type:law_presidential_republic
				has_variable = vice_president
				NOT = {
					any_scope_character = {
						has_variable = vice_president_trigger
					}
				}
			}
			random_scope_character = {
				limit = {
					is_ruler = no
					is_heir = no
					NOR = {
						has_role = general
						has_role = admiral
						has_role = agitator
						has_variable = vice_president_trigger
					}
					interest_group = {
						is_marginal = no
						ig_approval > 0
						ig_clout >= 0.1
					}
				}
				set_variable = vice_president_trigger
				save_scope_as = new_vice_president
				owner = {
					set_variable = {
						name = vice_president
						value = PREV
					}
				}
			}
			every_scope_character = {
				limit = {
					has_variable = vice_president_trigger
					NOT = { THIS = scope:new_vice_president }
				}
				remove_variable = vice_president_trigger
			}
		}	
		if = { #Removes vice president if no longer presidential republic
			limit = {
				has_variable = vice_president
				NOT = {
					has_law = law_type:law_presidential_republic
				}
			}
			remove_variable = vice_president
			random_scope_character = {
				limit = {
					has_variable = vice_president_trigger
				}
				remove_variable = vice_president_trigger
			}
		}
		if = { #Removes vice president if ruler
			limit = {
				ruler = { has_variable = vice_president_trigger }
			}
			ruler = { remove_variable = vice_president_trigger }
		}
	}
}

vfm_prime_ministers = {
	effect = {
		if = {
			limit = {
				OR = {
					has_law = law_type:law_monarchy
					has_law = law_type:law_parliamentary_republic
				}
				NOT = { has_law = law_type:law_autocracy }
				NOT = { has_variable = prime_minister }
			}
			random_scope_character = {
				limit = {
					is_ruler = no
					is_heir = no
					NOR = {
						has_role = general
						has_role = admiral
						has_role = agitator
						has_variable = prime_minister_trigger
					}
					interest_group = {
						is_marginal = no
						ig_approval > 0
						ig_clout >= 0.1
					}
				}
				set_variable = prime_minister_trigger
				owner = {
					set_variable = {
						name = prime_minister
						value = PREV
					}
				}
			}
		}
		if = {
			limit = {
				OR = {
					has_law = law_type:law_monarchy
					has_law = law_type:law_parliamentary_republic
				}
				NOT = { has_law = law_type:law_autocracy }
				has_variable = prime_minister
				NOT = { 
					any_scope_character = {
						has_variable = prime_minister_trigger
					}
				}
			}
			random_scope_character = {
				limit = {
					is_ruler = no
					is_heir = no
					NOR = {
						has_role = general
						has_role = admiral
						has_role = agitator
						has_variable = prime_minister_trigger
					}
					interest_group = {
						is_marginal = no
						ig_approval > 0
						ig_clout >= 0.1
					}
				}
				set_variable = prime_minister_trigger
				save_scope_as = new_prime_minister
				every_scope_character = {
					limit = {
						has_variable = prime_minister_trigger
						NOT = { THIS = scope:new_prime_minister }
					}
					remove_variable = prime_minister_trigger
				}
				owner = {
					set_variable = {
						name = prime_minister
						value = PREV
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {
					AND = {
						OR = {
							has_law = law_type:law_monarchy
							has_law = law_type:law_parliamentary_republic
						}
						NOT = { has_law = law_type:law_autocracy }
					}
				}
				has_variable = prime_minister
			}
			remove_variable = prime_minister
			random_scope_character = {
				limit = {
					has_variable = prime_minister_trigger
				}
				remove_variable = prime_minister_trigger
			}
		}
		if = { #Removes prime_minister if ruler
			limit = {
				ruler = { has_variable = prime_minister_trigger }
			}
			ruler = { remove_variable = prime_minister_trigger }
		}
	}
}
