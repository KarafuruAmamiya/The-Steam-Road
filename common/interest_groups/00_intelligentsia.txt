ig_intelligentsia = {
	color = hsv{ 0.13 0.64 0.99 }
	texture = "gfx/interface/icons/ig_icons/intelligensia.dds"
	layer = "revolution_dynamic_intelligentsia"
	index = 3

	ideologies = {
		ideology_nationalist
		ideology_liberal
		ideology_constitutionalist
		ideology_anti_slavery
	}

	enable = {
		always = yes
	}

	# Deprecated; use on_enable effect to assign traits instead
	traits = {
		ig_trait_propagandists
		ig_trait_avant_garde
		ig_trait_social_criticism
	}

	on_enable = {
		ig:ig_intelligentsia = {
			if = {
				limit = {
					owner = { c:CHI ?= this }
				}
				set_ig_trait = ig_trait:ig_trait_social_criticism
				set_ig_trait = ig_trait:ig_trait_avant_garde
				set_ig_trait = ig_trait:ig_trait_restoration
			}
			else_if = {
				limit = {
					owner = { 
						OR = { 
							c:TUR ?= this
							c:PER ?= this
						}
					}
				}
				set_ig_trait = ig_trait:ig_trait_crisis_of_identity
				set_ig_trait = ig_trait:ig_trait_avant_garde
				set_ig_trait = ig_trait:ig_trait_propagandists
			}
			else_if = {
				limit = {
					has_dlc_feature = voice_of_the_people_content
					owner = { c:FRA ?= this }
				}
				set_ig_trait = ig_trait:ig_trait_social_criticism
				set_ig_trait = ig_trait:ig_trait_avant_garde
				set_ig_trait = ig_trait:ig_trait_les_beaux_arts
			}
			else = {
				set_ig_trait = ig_trait:ig_trait_social_criticism
				set_ig_trait = ig_trait:ig_trait_avant_garde
				set_ig_trait = ig_trait:ig_trait_propagandists
			}
		}
		# Anti-Clerical Intelligentsia
		if = {
			limit = {
				NOR = {
					has_law = law_type:law_theocracy
					country_has_state_religion = rel:confucian
					NOT = { has_technology_researched = empiricism }
				}
			}
			ig:ig_intelligentsia = {
				add_ideology = ideology_anti_clerical
			}
		}
		# Literati
		else_if = {
			limit = {
				any_primary_culture = {
					has_discrimination_trait = han_chinese
				}
			}
			ig:ig_intelligentsia = { 
				set_interest_group_name = ig_literati
				add_ideology = ideology_westernizer
			}
		}
		# Gunpowder Empire Intelligentsia
		else_if = {
			limit = {
				OR = {
					c:TUR ?= this
					c:AKN ?= this
					c:PER ?= this
					is_country_type = unrecognized
					has_technology_researched = international_trade
				}
			}
			ig:ig_intelligentsia = {
				add_ideology = ideology_westernizer
			}
		}
		# Republican Intelligentsia
		else_if = {
			limit = {
				coa_republic_trigger = yes
			}
			ig:ig_intelligentsia = { 
				remove_ideology = ideology_constitutionalist
				add_ideology = ideology_republican
			}
		}
		# Republican Intelligentsia (Dutch)
		else_if = {
			limit = {
				c:NET ?= this
			}
			ig:ig_intelligentsia = { 
				remove_ideology = ideology_constitutionalist
				add_ideology = ideology_republican
			}
		}
		# Revolutionary Republican Intelligentsia
		else_if = {
			limit = {
				coa_republic_trigger = yes
				OR = {
					c:ITA ?= this
					is_subject_of = c:ITA
				}
			}
			ig:ig_intelligentsia = { 
				remove_ideology = ideology_constitutionalist
				remove_ideology = ideology_republican
				add_ideology = ideology_revolutionary_republican
			}
		}
		# New World Intelligentsia
		else_if = {
			limit = {
				country_is_in_new_word = yes
			}
			ig:ig_intelligentsia = { 
				remove_ideology = ideology_nationalist
				add_ideology = ideology_segregationist
			}
		}
		if = {
			limit = {
				has_technology_researched = feminism
			}
			every_interest_group = {
				limit = {
					has_ideology = ideology:ideology_liberal
				}
				remove_ideology = ideology_liberal
				add_ideology = ideology_liberal_modern
			}
		}
	}

			limit = {
				is_marginal = yes
			}
			multiply = {
				value = 0.1
			}
		}
		# Monarch is more likely to adopt a Powerful IG

			limit = {
				is_powerful = yes
			}
			multiply = {

			}
		}

			limit = {
				owner = {
					has_law = law_type:law_technocracy
				}
			}
			multiply = {
				value = 3
			}
		}

			limit = {

					has_law = law_type:law_appointed_bureaucrats

				value = 1.5
			}
		}
		# Impossible without Empiricism

			limit = {
				owner = {
					NOT = { has_technology_researched = empiricism }
