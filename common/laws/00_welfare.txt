law_no_social_security = {
	group = lawgroup_welfare
	
	icon = "gfx/interface/icons/law_icons/no_social_security.dds"
	
	progressiveness = 0
	
	possible_political_movements = {
		law_poor_laws
		law_wage_subsidies
	}
	
	pop_support = {
		value = 0
		multiply = {
			desc = "TUR_NO_BALKAN_REVOLTS_OVER_LAWS"			
			if = {
				limit = {
					owner = { 
						c:TUR ?= this
					}
					pop_is_discriminated = yes
				}
				value = 0
			}
		}
	}

	# AI will never abolish institutions entirely
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = {
				law_is_important = yes
			}
			add = 750
		}
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
}

		law_wage_subsidies
	}
	
	pop_support = {
		value = 0
		
		add = {
			desc = "POP_POVERTY"			
			if = {
				limit = {
					owner = { 
						NOR = { 
							has_law = law_type:law_old_age_pension

						} 
					}
				}		
			# SoL 14 gives 0.1 pop support, 13 -> 0.2, 12 -> 0.3, ... 5 -> 1.0

				max = 0.5
