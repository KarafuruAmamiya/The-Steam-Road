# HRE

je_hre_falls_apart = { 
	icon = "gfx/interface/icons/event_icons/event_hre.dds"
	group = je_group_tsr_content
	
	should_be_pinned_by_default = yes

	complete = {
		is_power_bloc_leader = yes
		power_bloc ?= {
			any_power_bloc_member = {
				custom_tooltip = {
					text = hre_complete_tt
					is_direct_subject_of = root
					NOT = { is_subject_type = subject_type_hre }
				}
				percent = 1
			}
			current_cohesion_number > 50
		}
	}
	
	fail = {
		OR = {
			is_power_bloc_leader = no
			custom_tooltip = {
				text = hre_fail_tt
				any_in_global_list = {
					variable = hre_member
					NOT = { is_in_same_power_bloc = root }
				}
			}
		}
	}
	
	timeout = 4380
	
	on_fail = {
		set_global_variable = hre_end
		c:SAX = { trigger_event = { id = hre.1 } }
		hidden_effect = {
			every_country = {
				limit = {
					OR = {
						country_is_in_hre = yes
						has_strategic_adjacency = c:SAX
					}
				}
				trigger_event = { id = hre.2 }
			}
			clear_global_variable_list = hre_member
		}
	}

	on_timeout = {
		set_global_variable = hre_end
		c:SAX = { trigger_event = { id = hre.1 } }
		hidden_effect = {
			every_country = {
				limit = {
					NOT = { c:SAX ?= this }
					OR = {
						country_is_in_hre = yes
						has_strategic_adjacency = c:SAX
					}
				}
				trigger_event = { id = hre.2 }
			}
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_hre_falls_apart_status
			}
		}
	}
	
	weight = 100
	
	invalid = {
		c:SAX = {
			NOT = { 
				is_power_bloc_leader = yes
				power_bloc ?= {
					has_identity = identity:identity_hre
				}
			}
		}
	}
	
	transferable = no
}

# Germany

je_reconquer_germany = { 
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_tsr_content
	
	should_be_pinned_by_default = yes

	complete = {
		any_state = {
			filter = {
				OR = {
					region = sr:region_north_germany
					region = sr:region_south_germany
					region = sr:region_rhine
				}
				state_region = {
					OR = {
						is_homeland = cu:north_german
						is_homeland = cu:south_german
						is_homeland = cu:rhenish
					}
				}
			}
			owner = root
			percent >= 0.8
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_reconquer_germany_war_status
			}
		}
	}
	
	weight = 100
	
	transferable = no
}
