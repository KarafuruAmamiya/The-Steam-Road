		}

		name = usc_events.1.a

		}

# USC Requests Debt Renegotiation
usc_events.3 = {
	type = country_event
	placement = root
	
	title = usc_events.3.t
	desc = usc_events.3.d
	flavor = usc_events.3.f
	
	event_image = {
		video = "africa_diplomats_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	
	duration = 3
	
	trigger = {
	}

	immediate = {
		c:USC = {
			save_scope_as = usc_scope
			set_variable = usc_debt_renegotiated_var
		}
		if = {
			limit = {
				exists = c:SWE
			}
			c:SWE = {
				save_scope_as = swe_scope
			}
		}
		else_if = {
			limit = {
				exists = c:SCA
			}
			c:SCA = {
				save_scope_as = swe_scope
			}
		}
	}

	option = {
		name = usc_events.3.a
		default_option = yes
		c:USC = { 
			trigger_event = { id = usc_events.4 popup = yes }
		}
		show_as_tooltip = {
			if = {
				limit = {
					scope:swe_scope = {
						owes_obligation_to = c:USC
					}
				}
				scope:swe_scope = {
					set_owes_obligation_to = {
						country = c:USC
						setting = no
					}
				}
			}
			else = {
				c:USC = {
					set_owes_obligation_to = {
						country = scope:swe_scope
						setting = yes
					}
				}
			}
		}
		if = {
			limit = {
				has_modifier = modifier_haitian_independence_payments_4
			}
			remove_modifier = modifier_haitian_independence_payments_4
			add_modifier = {
				name = modifier_haitian_independence_payments_3
				months = -1
			}
		}
		else_if = {
			limit = {
				has_modifier = modifier_haitian_independence_payments_3
			}
			remove_modifier = modifier_haitian_independence_payments_3
			add_modifier = {
				name = modifier_haitian_independence_payments_2
				months = -1
			}
		}
		else_if = {
			limit = {
				has_modifier = modifier_haitian_independence_payments_2
			}
			remove_modifier = modifier_haitian_independence_payments_2
			add_modifier = {
				name = modifier_haitian_independence_payments_1
				months = -1
			}
		}
		change_relations = {
			country = c:USC
			value = 20
		}
		ai_chance = {
			base = 75
		}
	}
	option = {
		name = usc_events.3.b
		c:USC = { trigger_event = { id = usc_events.5 popup = yes } }
		change_relations = {
			country = c:USC
			value = -20
		}
		ai_chance = {
			base = 25
		}
	}
}
