je_ruthenian_identity_emergence = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_tsr_content
	
	is_shown_when_inactive = {
		OR = {
			country_has_primary_culture = cu:ruthenian
			country_has_primary_culture = cu:cossack
		}
		hidden_trigger = {
			NOT = { has_variable = completed_culture_genesis }
		}
		hidden_trigger = {
			NOR = {
				c:GAL ?= this
				c:LIT ?= this
				c:RUT ?= this
			}
		}
		is_revolutionary = no
	}
	
	possible = {
		has_technology_researched = nationalism
		any_country = {
			filter = {
				OR = {
					country_has_primary_culture = cu:ruthenian
					country_has_primary_culture = cu:cossack				
				}
				hidden_trigger = {
					NOR = {
						c:GAL ?= this
						c:LIT ?= this
						c:RUT ?= this
						this = root
					}
				}
			}
			has_technology_researched = nationalism	
			percent >= 0.75
		}
		hidden_trigger = {
			NOT = { has_variable = completed_culture_genesis }
		}
	}

	immediate = {
		set_national_identity_vars = yes
		set_variable = { name = national_identity_progress_var value = 0 }
		calculate_national_identity_progress = yes
		if = {
			limit = {
				OR = {
					c:PLT ?= this
					c:SMO ?= this
				}
			}
			cu:ruthenian = {
				save_scope_as = old_national_culture
			}
			cu:byelorussian = {
				save_scope_as = new_national_culture
			}
		}
		if = {
			limit = {
				OR = {
					c:CHR ?= this
					c:ZAP ?= this
					c:SLB ?= this
					c:DON ?= this
				}
			}
			cu:ruthenian = {
				save_scope_as = old_national_culture
			}
			cu:ukrainian = {
				save_scope_as = new_national_culture
			}
		}
		trigger_event = { id = ruthenian_events.1 popup = yes }
	}

	on_weekly_pulse = {
		effect = {
			calculate_national_identity_progress = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			add_national_identity_progress = yes
		}
	}

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
	}

	on_complete = {
		clear_national_identity_vars = yes
		trigger_event = {
			id = ruthenian_events.2 popup = yes
		}
		show_as_tooltip = {
			if = {
				limit = {
					OR = {
						c:PLT ?= this
						c:SMO ?= this
					}
				}
				add_primary_culture = cu:byelorussian
				remove_primary_culture = cu:ruthenian
			}
			if = {
				limit = {
					c:CHR ?= this
				}
				add_primary_culture = cu:ukrainian
				remove_primary_culture = cu:ruthenian
			}
			if = {
				limit = {
					OR = {
						c:ZAP ?= this
						c:SLO ?= this
						c:DON ?= this
					}
				}
				add_primary_culture = cu:ukrainian
			}
		}
		hidden_effect = {
			set_variable = completed_culture_genesis
		}
	}

	fail = {
		custom_tooltip = {
			text = completed_pan_national_project_tt
			c:RUT ?= this
		}
	}
	
	on_fail = {
		hidden_effect = {
			set_variable = completed_culture_genesis
		}
	}

	current_value = {
		value = root.var:national_identity_progress_var
	}

	goal_add_value = {
		add = 100
	}

	weight = 100

	progressbar = yes

	should_be_pinned_by_default = yes
}

je_ukrainian_unification = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	
	group = je_group_tsr_content
	
	is_shown_when_inactive = {
		country_has_primary_culture = cu:ukrainian
		has_variable = ruthenian_culture_change
	}
	
	possible = {
		is_subject = no
		any_state = {
			state_region = {
				is_homeland = cu:ukrainian
			}
			count >= 5
		}
	}
	
	immediate = {
		hidden_effect = {
			set_strategy = ai_strategy_unify_ukraine
		}
		every_state = {
			limit = {
				state_region = {
					is_homeland = cu:ukrainian
				}
				NOT = { has_claim_by = root }
				NOT = { owner = root }
				NOT = { owner = { is_subject_of = root } }
			}
			state_region = {
				add_claim = root
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = ukraine_formation
			c:UKR ?= THIS
		}
	}

	on_complete = {
		#trigger_event = { id = ruthenian_events.6 popup = yes }
	}

	weight = 100
	should_be_pinned_by_default = yes
}
