# Balkan Crisis Success
balkan_crisis_2.3 = {
	type = country_event
	placement = root
	title = balkan_crisis_2.3.t
	desc = balkan_crisis_2.3.d
	flavor =  balkan_crisis_2.3.f
	
	duration = 3
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	
	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	immediate = {
		c:TUR ?= { save_scope_as = tur_scope }
	}

	option = {
		name = balkan_crisis_2.3.a
		default_option = yes
		
		add_modifier = {
			name = eternal_porte
			months = long_modifier_time
		}
		hidden_effect = {
			every_scope_state = {
				limit = {
					OR = {
						region = sr:region_greece
						region = sr:region_balkans
					}
				}
				add_modifier = {
					name = balkan_crisis_suppression
					months = long_modifier_time
				}
			}
		}
	}
}

		video = "middleeast_battlefield_trenches"

		}
		c:TUR ?= { save_scope_as = tur_scope }
	}

	option = {
		name = balkan_crisis_2.4.a
		default_option = yes
		trigger = { c:TUR ?= this }
		
		every_subject_or_below = {
			limit = { country_is_in_europe = yes }
			make_independent = yes
		}

		if = {
			limit = { exists = c:GRE }
			random_scope_state = {
				limit = { state_region = s:STATE_WESTERN_THRACE }
				set_state_owner = c:GRE
			}
		}

			limit = { NOT = { exists = c:MAC } }
			custom_tooltip = {
				text = macedonia_created_as_subject

							name = radicals_bandaid

balkan_crisis_2.7 = {
	type = country_event
	placement = root
	title = balkan_crisis_2.7.t
	desc = balkan_crisis_2.7.d
	flavor =  balkan_crisis_2.7.f
	
	duration = 3
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	
	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	immediate = {
		c:FRA ?= { save_scope_as = offer_scope_fra }
		random_scope_state = {
			limit = {
				state_region = s:STATE_CYPRUS
			}
			save_scope_as = cyprus_state
			set_variable = state_to_cede
		}
	}

	option = { # Accept the deal
		name = balkan_crisis_2.7.a
		
		custom_tooltip = cyprus_convention_tur_tt
		custom_tooltip = {
			text = balkan_crisis_progress_decrease_20_tt
			change_global_variable = { name = balkan_crisis_main_var add = -20 }
		}
		create_dynamic_country = {
			origin = ROOT
			tier = principality
			culture = cu:french
			religion = rel:orthodox
			
			capital = scope:cyprus_state
			cede_state_trigger = {
				has_variable = state_to_cede
			}
			
			on_created = {
				activate_law = law_type:law_governorate
				add_primary_culture = cu:greek
				add_primary_culture = cu:turkish
				every_state = {
					remove_variable = state_to_cede
				}
			}
		}
		hidden_effect = {
			c:FRA = {
				set_variable = cyprus_convention
				post_notification = tur_convention_accepted
				create_diplomatic_pact = {
					country = scope:cyprus_state.owner
					type = protectorate
				}
				set_secret_goal = {
					country = c:TUR
					secret_goal = reconcile
				}
			}
		}
		
		change_relations = {
			country = c:FRA
			value = 30
		}
		
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					OR = {
						c:FRA.army_size > root.army_size
						c:FRA.navy_size > root.navy_size
					}
				}
				add = 100
			}
		}
	}
	
	option = { #No
		name = balkan_crisis_2.7.b
		default_option = yes
		change_relations = {
			country = c:FRA
			value = -30
		}
		c:FRA = {
			post_notification = tur_convention_refused
		}
		
		ai_chance = {
			base = 10
		}
	}
}
