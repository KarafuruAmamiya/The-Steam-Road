pollit_constitution_button = {
	name = "pollit_constitution_button"
	desc = "pollit_constitution_button_desc"

	visible = { 
		always = yes
	}

	possible = {
		hidden_trigger = {
			OR = {
				has_global_variable = union_stall_1
				has_global_variable = union_stall_2
				has_global_variable = union_stall_3
				has_global_variable = union_stall_4
				has_global_variable = union_stall_5
			}
			NOT = { has_variable = constitution_button_done }
			NOT = { has_variable = checkpoint_ongoing }
		}
	}

	ai_chance = {
		base = 10
	}

	effect = {
		set_variable = constitution_button_done
		set_variable = checkpoint_ongoing
		add_journal_entry = { type = je_pollit_constitution }
	}
}

pollit_pledge_button = { #"Will you pledge an equal union with this nation, to be faithful to her as long as you both shall live?" "I do" /jk
	name = "pollit_pledge_button"
	desc = "pollit_pledge_button_desc"

	visible = { 
		always = yes
	}

	possible = {
		hidden_trigger = {
			has_global_variable = union_stall_6
			NOT = { has_variable = pledge_button_done }
			NOT = { has_variable = checkpoint_ongoing }
		}
		OR = {
			has_law = law_type:law_federal
			has_law = law_type:law_confederated
		}
		NOR = {
			has_law = law_type:law_ethnostate
			has_law = law_type:law_national_supremacy
		}
		any_scope_state = {
			filter = {
				state_region = {
					is_homeland = cu:polish
				}
			}
			radical_fraction = {
				value < 0.1
			}
			count = all
		}
		c:LIT = {
			relations:root >= relations_threshold:cordial
			any_scope_state = {
				filter = {
					state_region = {
						is_homeland = cu:lithuanian
					}
				}
				radical_fraction = {
					value < 0.1
				}
				count = all
			}
		}
	}

	ai_chance = {
		base = 10
	}

	effect = {
		set_variable = pledge_button_done
		if = {
			limit = { NOT = { country_has_primary_culture = cu:lithuanian } }
			add_primary_culture = cu:lithuanian
		}
	}
}
