je_odrodzenie = {
	icon = "gfx/interface/icons/event_icons/event_flags/event_plc_flag.dds"
	
	group = je_group_tsr_content
	
	scripted_button = je_odrodzenie_polish_radicals_subject_button
	scripted_button = je_odrodzenie_polish_radicals_overlord_button

	immediate = {
		root = { save_scope_as = poland_scope }
		trigger_event = { id = poland_lithuania_events.4 days = 3 popup = yes }
	}

	complete = {
		any_state = {
			filter = { 
				state_region = {
					region = sr:region_poland
					OR = {
						is_homeland = cu:polish
						is_homeland = cu:lithuanian
					}
				}
			}
			count = all
			owner = root
		}
	}

	on_complete = {
		set_variable = odrodzenie_complete
		trigger_event = { id = poland_events.4 popup = yes }
	}
	
	fail = {
		is_subject = yes
	}
	
	invalid = {
		NOT = { 
			exists = c:PLC 
		}
	}

	weight = 100

	should_be_pinned_by_default = yes
}

je_union_of_augustow_pol = {
	icon = "gfx/interface/icons/event_icons/event_plc.dds"
	
	group = je_group_tsr_content
	
	scripted_button = pollit_constitution_button
	scripted_button = pollit_industrialization_button
	scripted_button = pollit_defense_button
	scripted_button = pollit_sovereignty_button
	scripted_button = pollit_nationalism_button
	scripted_button = pollit_pledge_button
	
	immediate = {
		set_plc_point_vars = yes
		c:POL ?= {
			save_scope_as = pol_scope
		}
		c:LIT ?= {
			save_scope_as = lit_scope
		}
    }
	
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_global_variable = plc_point_freeze
					}
				}
				calculate_plc_point = yes
			}
		}
	}

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		c:LIT ?= {
			is_subject_of = root
			relations:root >= relations_threshold:cordial
			has_attitude = {
				who = root
				attitude = loyal
			}
		}
		#In SOI, expand to lower Lithuania's liberty desire
	}

	on_complete = {
		trigger_event = { id = poland_lithuania_events.5 popup = yes }
		show_as_tooltip = {
			annex_with_incorporation = c:LIT
			change_tag = PLC
			add_journal_entry = { type = je_odrodzenie }
		}
	}
	
	fail = {
		OR = {
			custom_tooltip = {
				text = plc_union_fail
				global_var:plc_union_progress_var <= 0
			}
			is_subject = yes
			c:LIT ?= { NOT = { is_subject_of = root } }
			has_war_with = c:LIT
		}
	}
	
	on_fail = {
		set_global_variable = plc_union_fail_var
		trigger_event = { id = poland_lithuania_events.6 popup = yes }
	}
	
	invalid = {
		NOT = { exists = c:LIT }
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_global_variable = plc_point_freeze
					}
				}
				calculate_plc_point = yes
				add_plc_point = yes
			}
			if = {
				limit = {
					OR = {
						any_civil_war = {
							civil_war_progress > 0.5
						}
						c:LIT = {
							any_civil_war = {
								civil_war_progress > 0.5
							}
						}
					}
					NOT = { has_global_variable = plc_civilwar_point }
				}
				change_global_variable = { name = plc_union_progress_var subtract = 10 }
				set_global_variable = { name = plc_civilwar_point years = 3 }
			}
			if = {
				limit = {
					OR = {
						in_default = yes
						has_modifier = declared_bankruptcy
					}
					NOT = { has_global_variable = plc_default_point }
				}
				change_global_variable = { name = plc_union_progress_var subtract = 10 }
				set_global_variable = { name = plc_default_point }
			}
			if = {
				limit = {
					country_rank = rank_value:major_power
					is_losing_power_rank = no
					NOT = { has_global_variable = plc_regional_power_point }
				}
				change_global_variable = { name = plc_union_progress_var add = 10 }
				set_global_variable = { name = plc_regional_power_point }
			}
			if = {
				limit = {
					country_rank = rank_value:great_power
					is_losing_power_rank = no
					NOT = { has_global_variable = plc_great_power_point }
				}
				change_global_variable = { name = plc_union_progress_var add = 10 }
				set_global_variable = { name = plc_great_power_point }
			}
			
			#Checkpoints
			if = {
				limit = {
					global_var:plc_union_progress_var >= 50
					NOT = { has_global_variable = union_stall_1 }
				}
				set_global_variable = union_stall_1
				freeze_plc_point = yes
			}
			if = {
				limit = {
					global_var:plc_union_progress_var >= 100
					NOT = { has_global_variable = union_stall_2 }
				}
				set_global_variable = union_stall_2
				freeze_plc_point = yes
				if = {
					limit = { has_variable = checkpoint_ongoing }
					remove_variable = checkpoint_ongoing
				}
			}
			if = {
				limit = {
					global_var:plc_union_progress_var >= 150
					NOT = { has_global_variable = union_stall_3 }
				}
				set_global_variable = union_stall_3
				freeze_plc_point = yes
				if = {
					limit = { has_variable = checkpoint_ongoing }
					remove_variable = checkpoint_ongoing
				}
			}
			if = {
				limit = {
					global_var:plc_union_progress_var >= 200
					NOT = { has_global_variable = union_stall_4 }
				}
				set_global_variable = union_stall_4
				freeze_plc_point = yes
				if = {
					limit = { has_variable = checkpoint_ongoing }
					remove_variable = checkpoint_ongoing
				}
			}
			if = {
				limit = {
					global_var:plc_union_progress_var >= 250
					NOT = { has_global_variable = union_stall_5 }
				}
				set_global_variable = union_stall_5
				freeze_plc_point = yes
				if = {
					limit = { has_variable = checkpoint_ongoing }
					remove_variable = checkpoint_ongoing
				}
			}
			if = {
				limit = {
					global_var:plc_union_progress_var >= 300
					NOT = { has_global_variable = union_stall_6 }
				}
				set_global_variable = union_stall_6
				freeze_plc_point = yes
				if = {
					limit = { has_variable = checkpoint_ongoing }
					remove_variable = checkpoint_ongoing
				}
			}
		}
		random_events = {
			100 = 0
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = union_progressing
				trigger = {
					global_var:plc_point_var_next > 0
				}
			}
			triggered_desc = {
				desc = union_regressing
				trigger = {
					global_var:plc_point_var_next < 0
				}
			}
			triggered_desc = {
				desc = union_stall
				trigger = {
					global_var:plc_point_var_next = 0
				}
			}
		}
	}
	
	current_value = {
		value = global_var:plc_union_progress_var
	}
	
	goal_add_value = {
		add = 360
	}
	
	progressbar = yes

	weight = 10000
	should_be_pinned_by_default = yes
}
