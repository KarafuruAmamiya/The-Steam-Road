je_usc_debt_negotiation_button = {
	name = "je_usc_debt_negotiation_button"
	desc = "je_usc_debt_negotiation_button_desc"

	visible = { NOT = { has_variable = us_debt_renegotiated_var } }

	ai_chance = {
		base = 10
	}

	possible = {
		trigger_if = {
			limit = { exists = c:SWE }
			NOT = { 
				has_war_with = c:SWE
				owes_obligation_to = c:SWE 
			}
			OR = {
				c:SWE = {
					relations:root >= relations_threshold:friendly
				}
				is_subject_of = c:SWE
				has_diplomatic_pact = {
					type = alliance
					who = c:SWE
				}
			}
		}
		
		trigger_else_if = {
			limit = { exists = c:SCA }
			NOT = { 
				has_war_with = c:SCA
				owes_obligation_to = c:SCA 
			}
			OR = {
				c:SCA = {
					relations:root >= relations_threshold:friendly
				}
				is_subject_of = c:SCA
				has_diplomatic_pact = {
					type = alliance
					who = c:SCA
				}
			}
		}
		
		OR = {
			has_modifier = modifier_french_independence_payments_1
			has_modifier = modifier_french_independence_payments_2
			has_modifier = modifier_french_independence_payments_3
			has_modifier = modifier_french_independence_payments_4
		}
	}

	effect = {
		if = {
			limit = { exists = c:SWE }
			c:SWE = { trigger_event = { id = usc_events.3 popup = yes } }
		}
		else_if = {
			limit = { exists = c:SCA }
			c:SCA = { trigger_event = { id = usc_events.3 popup = yes } }
		}
	}
}
