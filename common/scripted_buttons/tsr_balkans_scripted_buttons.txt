balkan_crisis_deescalate_france = {
	name = "balkan_crisis_deescalate_france"
	desc = "balkan_crisis_deescalate_france_desc"

	visible = { 
		c:FRA ?= this
		NOT = { has_variable = balkan_war_attacker }
	}

	possible = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		custom_tooltip = {
			text = balkan_crisis_50_tt
			global_var:balkan_crisis_main_var >= 50
		}
		hidden_trigger = {
			NOT = { has_variable = offer_cyprus_convention }
		}
	}

	effect = {
		custom_tooltip = {
			text = cyprus_convention_tt
			set_variable = offer_cyprus_convention
		}
		c:TUR ?= { trigger_event = { id = balkan_crisis_2.7 } }
	}
	
	ai_chance = {
		base = 0
		modifier = {
			trigger = {
				any_country = {
					has_variable = balkan_war_attacker
					NOT = { this = root }
					relations:root < relations_threshold:neutral
				}
			}
			add = 10
		}
		modifier = {
			trigger = {
				has_strategy = ai_strategy_maintain_power_balance 
			}
			add = 10
		}
		modifier = {
			trigger = {
				OR = {
					has_attitude = {
						who = c:TUR
						attitude = protective
					}
					has_attitude = {
						who = c:TUR
						attitude = befriend
					}
					has_attitude = {
						who = c:TUR
						attitude = conciliatory
					}						
				}
			}
			add = 10
		}
	}
}
