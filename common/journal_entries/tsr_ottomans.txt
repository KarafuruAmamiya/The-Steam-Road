			c:AKN ?= {
				is_subject_of = ROOT
			}
			c:KUR ?= {

			}
			AND = {
				country_or_subject_owns_entire_state_region = STATE_HARPUT
				country_or_subject_owns_entire_state_region = STATE_DIYARBAKIR
				country_or_subject_owns_entire_state_region = STATE_TRABZON

			}
		}

		trigger_event = { id = treaty_of_baglar_events.6 popup = yes }
	}

	immediate = {
		trigger_event = { id = treaty_of_baglar_events.4 popup = yes }
