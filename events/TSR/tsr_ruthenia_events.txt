# Birth of a National Identity
ruthenian_events.2 = {
	type = country_event
	placement = ROOT
	
	title = ruthenian_events.2.t
	desc = ruthenian_events.2.d
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = ruthenian_events.2.f2
				trigger = {
					OR = {
						c:PLT ?= this
						c:SMO ?= this
					}
				}
			}
			triggered_desc = {
				desc = ruthenian_events.2.f1
				trigger = {
					OR = {
						c:CHR ?= this
						c:ZAP ?= this
						c:SLB ?= this
						c:DON ?= this
					}
				}
			}
		}
	}
	
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	
	duration = 3
	
	trigger = {
	}

	immediate = {
		cu:ruthenian = {
			save_scope_as = old_ruthenian_culture
		}
		if = {
			limit = {
				OR = {
					c:PLT ?= this
					c:SMO ?= this
				}
			}
			cu:byelorussian = { save_scope_as = new_national_culture }
		}
		else_if = {
			limit = {
				OR = {
					c:CHR ?= this
					c:ZAP ?= this
					c:SLB ?= this
					c:DON ?= this
				}
			}
			cu:ukrainian = { save_scope_as = new_national_culture }
		}
		root = { save_scope_as = ruthenian_nation_scope }
	}

	option = {
		name = ruthenian_events.2.a
		default_option = yes
		ai_chance = {
			base = 90
		}

		if = {
			limit = {
				OR = {
					c:PLT ?= this
					c:SMO ?= this
				}
			}
			add_primary_culture = cu:byelorussian
			remove_primary_culture = cu:ruthenian
			capital.state_region = {
				add_homeland = cu:byelorussian
				remove_homeland = cu:ruthenian
			}
		}
		if = {
			limit = {
				c:CHR ?= this
			}
			add_primary_culture = cu:ukrainian
			remove_primary_culture = cu:ruthenian
			capital.state_region = {
				add_homeland = cu:ukrainian
				remove_homeland = cu:ruthenian
			}
		}
		if = {
			limit = {
				OR = {
					c:ZAP ?= this
					c:SLB ?= this
					c:DON ?= this
				}
			}
			add_primary_culture = cu:ukrainian
			capital.state_region = {
				add_homeland = cu:ukrainian
				remove_homeland = cu:ruthenian
			}
		}
		add_journal_entry = {
			type = je_ukraine_belarus_national_identity
		}
		every_country = {
			limit = {
				NOT = {
					THIS = ROOT
				}
				has_interest_marker_in_region = root.capital.region
			}
			post_notification = ruthenian_culture_genesis
		}
	}
}

ruthenian_events.3 = {
	type = country_event
	placement = ROOT
	
	title = ruthenian_events.3.t
	desc = ruthenian_events.3.d
	flavor = ruthenian_events.3.f
	
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	
	duration = 3

	immediate = {
		set_variable = ruthenian_culture_change
		cu:ruthenian = {
			save_scope_as = old_national_culture
		}
		if = {
			limit = {
				OR = {
					c:PLT ?= this
					c:SMO ?= this
				}
			}
			cu:byelorussian = { save_scope_as = new_national_culture }
		}
		else_if = {
			limit = {
				OR = {
					c:CHR ?= this
					c:ZAP ?= this
					c:SLB ?= this
					c:DON ?= this
				}
			}
			cu:ukrainian = { save_scope_as = new_national_culture }
		}
	}

	option = {
		name = ruthenian_events.3.a
		default_option = yes
		custom_tooltip = {
			text = every_pop_converts_tt
			every_scope_pop = {
				limit = {
					culture = scope:old_national_culture
				}
				change_pop_culture = {
					target = scope:new_national_culture
					value = 1
				}
			}
		}
		
		hidden_effect = {
			random_country = {
				limit = {
					any_scope_state = {
						any_scope_pop = {
							culture = cu:ruthenian
						}
					}
					OR = {
						NOR = { 
							country_has_primary_culture = cu:ruthenian
							country_has_primary_culture = cu:cossack
						}
						c:LIT ?= this
						c:GAL ?= this
					}
					NOT = { has_variable = ruthenian_culture_change }
				}
				trigger_event = { id = ruthenian_events.5 }
			}
		}
	}
}

ruthenian_events.4 = {
	type = country_event
	placement = scope:ruthenian_culture_state
	
	title = ruthenian_events.4.t
	desc = ruthenian_events.4.d
	flavor = ruthenian_events.4.f
	
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	
	duration = 3
	
	trigger = {
	}

	immediate = {
		random_scope_state = {
			limit = {
				is_incorporated = yes
				state_region = {
					is_homeland = scope:old_national_culture
				}
				any_neighbouring_state = {
					state_region = {
						is_homeland = scope:new_national_culture
					}
					OR = {
						is_capital = yes
						custom_tooltip = {
							text = new_culture_has_25_percent_tt
							culture_percent_state = {
								target = scope:new_national_culture
								value >= 0.25
							}
						}
					}
				}
				OR = {
					any_scope_building = {
						OR = {
							is_building_type = building_naval_base
							is_building_type = building_barracks
						}
						level >= 15
					}
					any_scope_building = {
						is_building_type = building_university
						level >= 5
					}
					any_scope_building = {
						is_building_type = building_government_administration
						level >= 5
					}
					any_scope_building = {
						is_building_type = building_urban_center
						level >= 5
					}
				}
			}
			random_neighbouring_state = {
				limit = {
					state_region = {
						is_homeland = scope:new_national_culture
					}
					OR = {
						is_capital = yes
						custom_tooltip = {
							text = new_culture_has_25_percent_tt
							culture_percent_state = {
								target = scope:new_national_culture
								value >= 0.25
							}
						}
					}
					any_neighbouring_state = {
						state_region = {
							is_homeland = scope:old_national_culture
						}
					}
				}
				save_scope_as = ruthenian_culture_state
			}
		}
	}

	option = {
		name = ruthenian_events.4.a
		default_option = yes
		scope:ruthenian_culture_state = {
			if = {
				limit = {
					any_neighbouring_state = {
						owner = root
						state_region = {
							is_homeland = scope:old_national_culture
							NOT = {
								is_homeland = scope:new_national_culture
							}
						}
					}
				}
				every_neighbouring_state = {
					limit = {
						owner = root
						state_region = {
							is_homeland = scope:old_national_culture
							NOT = {
								is_homeland = scope:new_national_culture
							}
						}
						is_incorporated = yes
						OR = {
							any_scope_building = {
								OR = {
									is_building_type = building_naval_base
									is_building_type = building_barracks
								}
								level >= 15
							}
							any_scope_building = {
								is_building_type = building_university
								level >= 5
							}
							any_scope_building = {
								is_building_type = building_government_administration
								level >= 5
							}
							any_scope_building = {
								is_building_type = building_urban_center
								level >= 5
							}
						}
					}
					state_region = {
						if = {
							limit = {
								OR = {
									c:PLT ?= root
									c:SMO ?= root
								}
							}
							remove_homeland = cu:ruthenian
							add_homeland = cu:byelorussian
						}
						if = {
							limit = {
								OR = {
									c:CHR ?= root
									c:ZAP ?= root
									c:SLB ?= root
									c:DON ?= root
								}
							}
							remove_homeland = cu:ruthenian
							add_homeland = cu:ukrainian
						}
					}
				}
			}
			else = { #For islands and such
				random_scope_state = {
					limit = {
						owner = root
						state_region = {
							is_homeland = scope:old_national_culture
							NOT = {
								is_homeland = scope:new_national_culture
							}
						}
						is_incorporated = yes
						OR = {
							any_scope_building = {
								OR = {
									is_building_type = building_naval_base
									is_building_type = building_barracks
								}
								level >= 15
							}
							any_scope_building = {
								is_building_type = building_university
								level >= 5
							}
							any_scope_building = {
								is_building_type = building_government_administration
								level >= 5
							}
							any_scope_building = {
								is_building_type = building_urban_center
								level >= 5
							}
						}
					}
					state_region = {
						if = {
							limit = {
