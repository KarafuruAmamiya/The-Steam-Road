je_russian_nationalism = {
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_tsr_content
	
    immediate = {
		hidden_effect = {
			set_variable = {
				name = russian_unrest_je_var
				value = 0
			}
		}
		set_russian_unrest_vars = yes
    }
	
	on_weekly_pulse = {
		effect = {
			calculate_russian_unrest = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			calculate_russian_unrest = yes
			add_russian_unrest = yes
		}
	}
	
    complete = {
		OR = {
			custom_tooltip = {
				text = russian_unrest_variable_at_full_tt
				root.var:russian_unrest_je_var >= 100
			}
			cu:russian = {
				culture_secession_progress = {
					target = ROOT
					value >= 0.9
				}
			}
		}
    }
	
    on_complete = {
        trigger_event = { id = russian_events.1 popup = yes }
    }

	current_value = {
		value = root.var:russian_unrest_je_var
	}
	
	fail = {
		root.average_sol > 10
		AND = {
			OR = {
				has_law = law_type:law_shared_heritage
				has_law = law_type:law_cultural_exclusion
				has_law = law_type:law_multicultural
			}
			OR = {
				has_law = law_type:law_freedom_of_conscience
				has_law = law_type:law_total_separation
				has_law = law_type:law_state_atheism
			}
			OR = {
				has_law = law_type:law_devolved
				has_law = law_type:law_federal
				has_law = law_type:law_confederated
			}
		}
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = cu:russian
				}
			}
			turmoil < 0.10
		}
		OR = {
			overlord = {
				AND = {
					OR = {
						has_law = law_type:law_shared_heritage
						has_law = law_type:law_cultural_exclusion
						has_law = law_type:law_multicultural
					}
					OR = {
						has_law = law_type:law_freedom_of_conscience
						has_law = law_type:law_total_separation
						has_law = law_type:law_state_atheism
					}
					OR = {
						has_law = law_type:law_devolved
						has_law = law_type:law_federal
						has_law = law_type:law_confederated
					}
				}
			}
			NOR = {
				is_subject_type = subject_type_dependency
				is_subject_type = subject_type_puppet
			}
		}
	}

	goal_add_value = {
		add = 100
	}
	
	invalid = {
		is_subject = no
	}
	
	on_invalid = {
		remove_primary_culture = cu:swedish
		add_primary_culture = cu:russian
		every_scope_character = {
			limit = {
				NOT = { culture = cu:russian }
			}
			kill_character = yes
		}
	}

	progressbar = yes

    weight = 1000
    should_be_pinned_by_default = yes
}

je_russian_independence_war = {
    icon = "gfx/interface/icons/event_icons/event_revolt.dds"

	group = je_group_tsr_content

    complete = {
		is_at_war = no
        is_subject = no
    }
    on_complete = {
		hidden_effect = {
			set_immune_to_revolutions = no
		}
        trigger_event = { id = russian_events.4 popup = yes }

		}

	on_fail = {
		hidden_effect = {
			set_immune_to_revolutions = no
		}
		trigger_event = { id = russian_events.5 popup = yes }
	}

    weight = 1000
    should_be_pinned_by_default = yes
}

je_moscow_uprising = {
    icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_tsr_content

    complete = {
		is_at_war = no
        c:MSC = {
			OR = {
				is_subject_of = root 
				is_country_alive = no
			}
        }
    }
    on_complete = {
		if = {
			limit = {
				exists = c:MSC
			}
			c:MSC = {
				hidden_effect = {
					set_immune_to_revolutions = no
				}
			}
		}
        trigger_event = { id = russian_events.5 popup = yes }
    }
    fail = {
		is_at_war = no
        trigger_if = {
            limit = {
                exists = c:MSC
            }
            NOT = {
                c:MSC = {
                    is_subject_of = root
                }
            }
        }
		trigger_else = {
            NOT = { owns_entire_state_region = STATE_MOSCOW }
        }
    }
	on_fail = {
		if = {
			limit = {
				exists = c:MSC
			}
			c:MSC = {
				hidden_effect = {
					set_immune_to_revolutions = no
				}
			}
		}
		trigger_event = { id = russian_events.4 popup = yes }
	}

    weight = 1000
    should_be_pinned_by_default = yes
}
