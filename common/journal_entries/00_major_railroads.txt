je_trans_siberian_railway = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		is_player = yes
		NOT = { has_global_variable = trans_siberian_done }
		owns_entire_state_region = STATE_UZBEKIA
		sr:region_east_siberia = {
			any_scope_state = {
				percent > 0.8
				owner = ROOT
			}
		}
		sr:region_west_siberia = {
			any_scope_state = {
				percent > 0.8
				owner = ROOT
			}
		}
	}

	possible = {
		has_technology_researched = railways
	}

	complete = {
		s:STATE_UZBEKIA = {
			any_scope_state = {
				owner = root
				has_building = building_railway
			}
		}
		s:STATE_SYRDARYA = {

			}

		}

		}
		s:STATE_TOMSK = {
			any_scope_state = {
