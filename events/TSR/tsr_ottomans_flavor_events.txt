##Turkish Nationalism

turkish_nationalism.1 = {
	type = country_event

	title = turkish_nationalism.1.t
	desc = turkish_nationalism.1.d
	flavor = turkish_nationalism.1.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	trigger = {
	 	#Triggered by journal entry
	}

	immediate = {
		random_scope_character = {
			limit = {
				AND = {
					OR = {
						has_role = agitator
						has_role = politician
					}
					OR = {
						has_ideology = ideology:ideology_turkish_nationalist
						has_ideology = ideology:ideology_radical
					}
				}
			}
			save_scope_as = nationalist_scope
			interest_group = {
				save_scope_as = nationalist_ig
			}
			set_variable = { name = memento_mori_immunity years = 20 }
		}
	}

	option = { 
		name = turkish_nationalism.1.a
		# nationalist movements will become entrenched
		scope:nationalist_ig = {
			add_modifier = {
				name = spreading_nationalist_sentiments
				months = long_modifier_time
			}
		}
		scope:nationalist_scope = {
			add_modifier = {
				name = popular_nationalist
			}
		}
	}
	option = { 
		name = turkish_nationalism.1.b
		default_option = yes
		trigger = {
			OR = {
				has_law = law_type:law_outlawed_dissent
				has_law = law_type:law_censorship
			}
		}
		scope:nationalist_ig = {
			add_modifier = {
				name = nationalists_suppressed
				months = long_modifier_time
			}
		}
		add_radicals = {
			value = medium_radicals
			interest_group = scope:nationalist_ig
		}
		scope:nationalist_scope = {
			add_modifier = {
				name = less_popular_nationalist_red
			}
		}
	}
}

turkish_nationalism.7 = {
	type = country_event
	placement = root
	event_image = {
		video = "europenorthamerica_springtime_of_nations"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	title = turkish_nationalism.7.t
	desc = turkish_nationalism.7.d
	flavor = turkish_nationalism.7.f

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_revolt.dds"

	duration = 3

	immediate = {
		random_country = {
			limit = {	
				is_revolutionary = yes
				country_definition = root.country_definition
				leader = {
					OR = {
						has_ideology = ideology:ideology_radical
						has_ideology = ideology:ideology_turkish_nationalist
					}
				}
			}
			save_scope_as = revolution_scope	
				
			#The Revolution shouldn't spawn in non-Turkish homelands
			scope:revolution_scope = {
				every_scope_state = {
					limit = { 
						state_region = {
							NOT = { is_homeland = cu:turkish }
						}
					}
					set_state_owner = root
				}
			}
			#The Revolution takes over Constantinople (or whatever your capital is)
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_EASTERN_THRACE
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_EASTERN_THRACE
					}
					set_state_owner = scope:target
				}
				scope:target ?= {
					every_scope_state = {
						set_state_type = incorporated
					}
				}
			}
			if = {
				limit = {
					scope:target ?= {
						any_scope_state = {
							state_region = s:STATE_EASTERN_THRACE
						}
					}
				}
				scope:target ?= {
					set_capital = STATE_EASTERN_THRACE
				}
			}
			else = {
				capital = {
					set_state_owner = scope:target
				}
				scope:target ?= {
					every_scope_state = {
						set_state_type = incorporated
					}
				}
			}
			
			if = {
				limit = {
					NOR = { 
						has_law = law_type:law_census_voting
						has_law = law_type:law_universal_suffrage
					}
				}
				activate_law = law_type:law_census_voting
			}
			if = {
				limit = {
					NOR = { 
						has_law = law_type:law_ethnostate
						has_law = law_type:law_national_supremacy 
					}
				}
				activate_law = law_type:law_national_supremacy
			}
			if = {
				limit = {
					NOT = { has_law = law_type:law_public_schools }
				}
				activate_law = law_type:law_public_schools
			}
			if = {
				limit = {
					NOR = { 
						has_law = law_type:law_freedom_of_conscience 
						has_law = law_type:law_shared_roots
					}
				}
				activate_law = law_type:law_freedom_of_conscience
			}
			if = {
				limit = {
					NOT = { has_law = law_type:law_slavery_banned }
				}
				activate_law = law_type:law_slavery_banned
			}
		}
	}

	option = {
		name = turkish_nationalism.7.a
		default_option = yes
		
		add_journal_entry = {
			type = je_turkish_revolutionary_war
			target = scope:revolution_scope
		}
	}
}
