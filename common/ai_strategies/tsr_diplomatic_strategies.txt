ai_strategy_unify_arabia = {
	icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"
	
	type = diplomatic
		
	undesirable_infamy_level = {
		value = 50
	}
	
	unacceptable_infamy_level = {
		value = 100
	}
	
	ideological_opinion_effect_mult = {
		value = 0.5
	}
	
	wargoal_maneuvers_fraction = {
		value = 0.1
	}		
	
	diplomatic_play_neutrality = {
		value = 0
	}
	
	diplomatic_play_boldness = {
		value = 50
	}
	
	# Added to value from default strategy
	obligation_value = {
		value = 0
	
		if = {
			limit = { 
				scope:target_country = { can_form_nation = ARA }
			}
			add = 25
		}
	}		
	
	recklessness = {
		value = 0.5
	}	
	
	aggression = {
		value = 0.5
		
		if = {
			limit = { 
				scope:target_country = { can_form_nation = ARA }
			}
			add = 10
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}					
			}
			multiply = 0.25
		}

		# More aggressive against ideological enemies, a bit less aggressive against ideological allies
		if = {
			limit = {
				"ai_ideological_opinion(scope:target_country)" < 0
			}
			
			multiply = {
				value = "ai_ideological_opinion(scope:target_country)"
				multiply = -0.2
				add = 1
				max = 4.0
			}				
		}	
		else = {
			multiply = {
				value = "ai_ideological_opinion(scope:target_country)"
				multiply = -0.05
				add = 1
				min = 0.5
			}			
		}		
	}

	building_group_weights = {
		bg_army = 1.5
	}
	
	wargoal_scores = {
		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA
					}
					can_reach_target_country = yes
				}
				add = 300
			}		
		}

		conquer_state = {
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA
						NOR = { 
							is_supporting_unification_candidate = { who = root country_formation = ARA }
							is_subject_of = root
							has_diplomatic_pact = {
								who = scope:target_country
								type = customs_union
							}							
						}
					}	
					is_adjacent_to_state = scope:target_state
				}
				add = 100
			}
			if = {
				limit = {
					scope:target_country = {
						is_unification_candidate = ARA
					}		
				}
				multiply = 0.0
			}
		}

		make_tributary = {	
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA					
						NOR = { 
							is_supporting_unification_candidate = { who = root country_formation = ARA }
							is_subject_of = root
							has_diplomatic_pact = {
								who = scope:target_country
								type = customs_union
							}							
						}	
					}
				}
				add = 200
			}	
			if = {
				limit = {
					is_unification_candidate = ARA
					scope:target_country = {
						is_unification_candidate = ARA
					}			
				}
				add = 100
			}	
		}

		return_state = {
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA					
					}
					is_adjacent_to_state = scope:target_state
				}

				add = 100
			}
			if = {
				limit = {
					scope:target_country = {
						is_unification_candidate = ARA
					}		
				}
				multiply = 0.0
			}
		}
		unification_leadership = {
			value = 0
		}		
		unification = {
			value = 1000
		}
	}			
	
	secret_goal_scores = {
		protect = {		
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA	
						NOT = { is_unification_candidate = ARA }						
						OR = { 
							is_supporting_unification_candidate = { who = root country_formation = ARA }
							is_subject_of = root
							has_diplomatic_pact = {
								who = scope:target_country
								type = customs_union
							}							
						}	
					}
				}
				add = 1000
			}		
		}		
		dominate = {		
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA
						NOR = { 
							is_supporting_unification_candidate = { who = root country_formation = ARA }
							is_subject_of = root
							has_diplomatic_pact = {
								who = scope:target_country
								type = customs_union
							}							
						}
					}
				}
				add = 1000
			}		
		}		
		conquer = {				
			if = {
				limit = {
					scope:target_country = {				
						can_form_nation = ARA
						NOT = {
							is_supporting_unification_candidate = { who = root country_formation = ARA }
						}
					}
				}
				add = 200
			}
			
			if = {
				limit = {
					scope:target_country = {
						NOT = {
							can_form_nation = ARA
						}
						any_scope_state = {
							is_homeland_of_country_cultures = c:ARA
						}
					}
				}
				add = 1000
			}
			
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = ARA
						is_subject = yes
					}
				}
				add = 1000
			}	
		}
		befriend = {
			if = {
				limit = {
					has_technology_researched = pan-nationalism
					scope:target_country = {
						can_form_nation = ARA
						country_rank < rank_value:minor_power
					}
				}
				add = 1000
			}
		}			
	}		
		
	possible = {
		is_unification_candidate = ARA	
		NOT = { country_tier = empire }
	}

	weight = {
		value = 50
		
		if = {
			limit = { NOT = { is_country_type = unrecognized } }
			add = 100
		}		
		
		if = {
			limit = { country_rank = rank_value:unrecognized_regional_power }
			add = 25
		}

		if = {
			limit = { country_rank = rank_value:unrecognized_major_power }
			add = 50
		}		
	}

	weight = {
		value = 1000	
	}
}

								is_homeland = cu:polish

				multiply = -0.2
				add = 1
				max = 4.0
			}				
		}	
		else = {
			multiply = {
				value = "ai_ideological_opinion(scope:target_country)"
				multiply = -0.05
				add = 1
				min = 0.5
			}			
		}			
	}

	building_group_weights = {
		bg_army = 1.5
	}
	
	secret_goal_weights = {
		protect = 0.5
		conquer = 1.5
	}

	wargoal_weights = {
		conquer_state = 1.5
		return_state = 1.5
		transfer_subject = 1.5
	}

	secret_goal_scores = {
		conquer = {
			if = {
				limit = {
					can_reach_target_country = yes
					has_strategic_land_adjacency = scope:target_country
					scope:target_country.country_rank <= root.country_rank
				}
			}
			if = {
				limit = { 
					scope:target_country = {
						any_scope_state = {
							state_region = {
								OR = {
									is_homeland = cu:polish
									is_homeland = cu:lithuanian
								}
							}
						}
					} 
				}
				add = 75
			}
		}
	}
		
	possible = {
		c:PLC ?= this
		has_journal_entry = je_odrodzenie
	}

	weight = {
		value = 50

		if = {
			limit = { ig:ig_armed_forces = { is_powerful = yes } }
			multiply = 1.5
		}
			
		if = {
			limit = { has_law = law_type:law_council_republic }

		}

		if = {
			limit = { country_rank = rank_value:great_power }
			add = 100
		}		
			
	}
}

			}
