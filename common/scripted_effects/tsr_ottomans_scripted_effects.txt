calculate_westernist_progress = {
	set_variable = { name = westernist_progress_var_next value = 0 }
	set_variable = { name = westernist_progress_from_radicals value = 0 }
	every_scope_state = {
		limit = {
			state_region = {
				is_homeland = cu:turkish
			}
			turmoil >= 0.10
		}
		ROOT = {
			change_variable = { name = westernist_progress_from_radicals add = prev.turmoil }
		}
	}
	set_variable = { name = westernist_progress_from_loyalists value = 0 }
	every_scope_state = {
		limit = {
			state_region = {
				is_homeland = cu:turkish
			}
			loyalty >= 0.10
		}
		ROOT = {
			change_variable = { name = westernist_progress_from_loyalists subtract = prev.loyalty }
		}
	}
	set_variable = { name = westernist_progress_from_characters value = 0 }
	every_scope_character = {
		limit = {
			has_ideology = ideology:ideology_westernist
		}
		ROOT = {
			change_variable = { name = westernist_progress_from_characters add = prev.popularity }
		}
	}
	change_variable = { name = westernist_progress_from_characters divide = 200 }
	# From Interest Groups
	set_variable = { name = westernist_progress_from_igs value = 0 }
	every_interest_group = {
		limit = {
			OR = {
				has_ideology = ideology:ideology_westernizer
				leader = {
					has_ideology = ideology:ideology_westernist
				}
			}
		}
		ROOT = {
			set_variable = { name = ig_weight value = prev.ig_clout }
		}
		if = {
			limit = {
				is_in_government = yes
			}
			ROOT = {
				change_variable = { name = ig_weight multiply = 3 }
			}
		}
		ROOT = {
			change_variable = { name = westernist_progress_from_igs add = var:ig_weight }
		}
	}
	# From Mehmed Domain
	set_variable = { name = westernist_progress_from_domain value = 0 }
	change_variable = {
		name = westernist_progress_from_domain 
		add = {
			value = 0.05
			multiply = num_mehmed_domain
		}
	}
	# From Legitimacy
	set_variable = { name = westernist_progress_from_legitimacy value = root.government_legitimacy }
	change_variable = { name = westernist_progress_from_legitimacy divide = 100 }
	change_variable = { name = westernist_progress_from_legitimacy subtract = 1 }
	change_variable = { name = westernist_progress_from_legitimacy multiply = -1 }

	# From Laws
	set_variable = { name = westernist_progress_from_laws value = 0 }
	# Free Speech
	set_variable = { name = temp_free_speech_var_contribution value = 0 }
	if = {
		limit = {
			has_law = law_type:law_outlawed_dissent
		}
		change_variable = { name = temp_free_speech_var_contribution add = -1 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_censorship
		}
		change_variable = { name = temp_free_speech_var_contribution add = -0.5 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_right_of_assembly
		}
		change_variable = { name = temp_free_speech_var_contribution add = 0.5 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_protected_speech
		}
		change_variable = { name = temp_free_speech_var_contribution add = 1 }
	}
	
	# Internal Affairs
	set_variable = { name = temp_ha_var_contribution value = institution:institution_home_affairs.investment }
	if = {
		limit = {
			has_law = law_type:law_no_home_affairs
		}
		change_variable = { name = temp_ha_var_contribution add = 0.3 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_guaranteed_liberties
		}
		change_variable = { name = temp_ha_var_contribution multiply = -0.1 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_national_guard
		}
		change_variable = { name = temp_ha_var_contribution multiply = -0.3 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_secret_police
		}
		change_variable = { name = temp_ha_var_contribution multiply = -0.5 }
	}
	change_variable = { name = westernist_progress_from_laws add = var:temp_free_speech_var_contribution }
	change_variable = { name = westernist_progress_from_laws add = var:temp_ha_var_contribution }
	remove_variable = temp_policing_var_contribution
	remove_variable = temp_ha_var_contribution
	
	# From Foreign Support
	set_variable = { name = westernist_progress_from_foreign_support value = 0 }
	change_variable = { 
		name = westernist_progress_from_foreign_support 
		add = {
			value = 0.1
			multiply = num_support_westernists
		}
	}
	change_variable = { 
		name = westernist_progress_from_foreign_support 
		subtract = {
			value = 0.1
			multiply = num_oppose_westernists
		}
	}
	
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_radicals }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_loyalists }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_characters }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_laws }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_igs }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_domain }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_legitimacy }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_from_foreign_support }
	change_variable = { name = westernist_progress_var_next add = var:westernist_progress_var_add }
}

set_westernist_vars = {
	set_variable = { name = westernist_progress_var_next value = 0 }
	set_variable = { name = westernist_progress_from_radicals value = 0 }
	set_variable = { name = westernist_progress_from_loyalists value = 0 }
	set_variable = { name = westernist_progress_from_laws value = 0 }
	set_variable = { name = westernist_progress_from_characters value = 0 }
	set_variable = { name = westernist_progress_from_igs value = 0 }
	set_variable = { name = westernist_progress_from_legitimacy value = 0 }
	set_variable = { name = westernist_progress_from_domain value = 0 }
	set_variable = { name = westernist_progress_from_foreign_support value = 0 }
	set_variable = { name = westernist_progress_var_add value = 0 }
}

clear_westernist_vars = {
	remove_variable = westernist_progress_var_next
	remove_variable = westernist_progress_from_radicals
	remove_variable = westernist_progress_from_loyalists
	remove_variable = westernist_progress_from_laws
	remove_variable = westernist_progress_from_characters
	remove_variable = westernist_progress_from_igs
	remove_variable = westernist_progress_from_domain
	remove_variable = westernist_progress_from_legitimacy
	remove_variable = westernist_progress_from_foreign_support
	remove_variable = westernist_progress_var_add
}

check_westernist_states = {
	if = {
		limit = { var:westernist_progress_var_next > 0 }
		if = {
			limit = {
				any_scope_state = {
					is_incorporated = yes
					state_region = {
						is_homeland = cu:turkish
					}
					is_capital = no
					NOT = { has_modifier = pro_mehmed_domain }
				}
			}
			random_scope_state = {
				limit = {
					is_incorporated = yes
					state_region = {
						is_homeland = cu:turkish
					}
					is_capital = no
					NOT = { has_modifier = pro_mehmed_domain }
				}
				save_scope_as = mehmed_defector_state
				add_modifier = {
					name = pro_mehmed_domain
				}
			}
			post_notification = westernist_state_defection
		}
	}
	else_if = {
		limit = { var:westernist_progress_var_next < 0 }
		if = {
			limit = {
				any_scope_state = {
					has_modifier = pro_mehmed_domain
				}
			}
			random_scope_state = {
				limit = {
					has_modifier = pro_mehmed_domain
				}
				save_scope_as = mehmed_return_state
				remove_modifier = pro_mehmed_domain
			}
			post_notification = westernist_state_return
		}
	}
}
