namespace = character_deaths

#President dies, Vice President rises
character_deaths.1 = {
	type = country_event
	placement = ROOT

	title = character_deaths.1.t
	desc = character_deaths.1.d
	flavor = character_deaths.1.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:former_vice_president

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 1
	
	immediate = {
		random_scope_character = {
			limit = {
				has_variable = vice_president_trigger
			}
			save_scope_as = former_vice_president
		}
	}

	option = {
		name = character_deaths.1.a
		default_option = yes
		scope:former_vice_president = {
			remove_variable = vice_president_trigger
			set_character_as_ruler = yes
		}
		hidden_effect = {
			random_scope_character = {
				limit = {
					is_ruler = no
					is_heir = no
					NOR = {
						has_role = general
						has_role = admiral
						has_role = agitator
						has_variable = vice_president_trigger
					}
					interest_group = {
						is_marginal = no
						ig_approval > 0
						ig_clout >= 0.1
					}
				}
				set_variable = vice_president_trigger
				save_scope_as = new_vice_president
				owner = {
					set_variable = {
						name = vice_president
						value = PREV
					}
				}
			}
			every_scope_character = {
				limit = {
					has_variable = vice_president_trigger
					NOT = { THIS = scope:new_vice_president }
				}
				remove_variable = vice_president_trigger
			}
		}
	}
}
