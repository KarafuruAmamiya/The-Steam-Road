tsr_test.2 = {
	type = country_event
	hidden = yes
	immediate = {	
		random_primary_culture = {
			save_scope_as = primary_culture
		}
		if = {
			limit = {
				any_scope_state = {
					state_population >= 100000
					is_split_state = no
					is_under_colonization = no
					state_region = {
						NOT = { is_homeland = scope:primary_culture }
					}
					culture_percent_state = {
						target = scope:primary_culture
						value >= 0.4
					}
				}
			}
			every_scope_state = {
				limit = {
					state_population >= 100000
					is_split_state = no
					is_under_colonization = no
					state_region = {
						NOT = { is_homeland = scope:primary_culture }
					}
					culture_percent_state = {
						target = scope:primary_culture
						value >= 0.4
					}
				}
				save_scope_as = new_homeland_state
			}
			scope:new_homeland_state = {
				state_region = {
					add_homeland = scope:primary_culture
				}
			}
			post_notification = new_homeland
		}
	}
}

tsr_test.4 = {
	type = country_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				has_law = law_type:law_governorate
			}
			if = {
				limit = {
					is_country_type = colonial
				}
				activate_law = law_type:law_presidential_republic
			}
			else = {
				enact_preferred_governance_principle_governorate = yes
			}
			restore_primary_culture = yes
			if = {
				limit = {
					has_variable = primary_culture_ruler
				}
				remove_variable = primary_culture_ruler
				ruler = {
					kill_character = yes
					hidden = yes
				}
				every_interest_group = {
					leader = {
						kill_character = yes
						hidden = yes
					}
				}
			}
		}
	}
}
